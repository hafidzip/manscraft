import { packChunk } from '../core/cellKey';
import { minImageF } from '../core/constants';
import type { TurretTarget } from './Turret';

/** Any source of turret targets. Lets the engine swap in a better index later. */
export interface TargetIndex {
  forEachNear(x: number, z: number, radius: number,
              cb: (t: TurretTarget) => void): void;
}

const CELL_LOG2 = 4;   // 16-block cells, reuses packChunk's key layout

/**
 * Rebuilt once per frame from the enemy list. Bucket arrays are reused, so a steady
 * state costs zero allocations. Cost is O(enemies) per frame instead of
 * O(guns x enemies) per acquire.
 */
export class EnemyGrid implements TargetIndex {
  readonly stats = { rebuilds: 0, targets: 0, queries: 0, cells: 0, candidates: 0 };
  private cells = new Map<number, TurretTarget[]>();
  private used: TurretTarget[][] = [];

  rebuild(list: Iterable<TurretTarget>): void {
    for (let i = 0; i < this.used.length; i++) this.used[i].length = 0;
    this.used.length = 0;
    let n = 0;
    for (const t of list) {
      if (!t.alive || t.cfg.peaceful) continue;   // same filter as the old acquire
      const k = packChunk(
        Math.floor(t.pos.x) >> CELL_LOG2,
        Math.floor(t.pos.z) >> CELL_LOG2,
      );
      let b = this.cells.get(k);
      if (!b) { b = []; this.cells.set(k, b); }
      if (b.length === 0) this.used.push(b);
      b.push(t);
      n++;
    }
    this.stats.rebuilds++;
    this.stats.targets = n;
  }

  forEachNear(x: number, z: number, radius: number,
              cb: (t: TurretTarget) => void): void {
    this.stats.queries++;
    const r = (Math.ceil(radius) + 15) >> CELL_LOG2;
    const cx = Math.floor(x) >> CELL_LOG2;
    const cz = Math.floor(z) >> CELL_LOG2;
    for (let dx = -r; dx <= r; dx++) {
      for (let dz = -r; dz <= r; dz++) {
        const b = this.cells.get(packChunk(cx + dx, cz + dz));
        if (!b || b.length === 0) continue;
        this.stats.cells++;
        for (let i = 0; i < b.length; i++) { this.stats.candidates++; cb(b[i]); }
      }
    }
  }
}
