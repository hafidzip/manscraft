      if (wy - sy > 3 || sy - wy > 3) return false;   // cave / bridge: not our layer
      if (c <= ARRIVE_COST) return false;             // melee steering owns this range
  
      // Bilinear blend of the four nearest cell-centre gradients. Without this
      // the direction snaps at every cell border and the crowd jitters.
      const fx = wx - bx - 0.5, fz = wz - bz - 0.5;
      const i0x = fx < 0 ? bx - 1 : bx, i0z = fz < 0 ? bz - 1 : bz;
      const tx  = fx < 0 ? fx + 1 : fx, tz  = fz < 0 ? fz + 1 : fz;
  
      let gx = 0, gz = 0, wsum = 0;
      for (let k = 0; k < 4; k++) {
        const ox = k & 1, oz = k >> 1;
        const w = (ox ? tx : 1 - tx) * (oz ? tz : 1 - tz);
        if (w <= 0) continue;
        if (!this.gradAt((i0x + ox) & MASK, (i0z + oz) & MASK)) continue;
        gx += this.gx * w; gz += this.gz * w; wsum += w;
      }
      if (wsum <= 0) return false;
  
      const m = Math.sqrt(gx * gx + gz * gz);
      if (m < 1e-4) return false;                     // plateau / local minimum
  
      out.x = gx / m;
      out.z = gz / m;
  
      const sx = out.x > 0.4 ? 1 : out.x < -0.4 ? -1 : 0;
      const sz = out.z > 0.4 ? 1 : out.z < -0.4 ? -1 : 0;
      const ti = ((bx + sx) & MASK) + ((((bz + sz) & MASK)) << SHIFT);
      const ty = this.surfY[ti];
      out.targetY = ty;
      out.climb = ty === NO_SURFACE ? 0 : ty - sy;
      out.cost = c / Q;
      return true;
    }
  
    /** Descent direction at a cell centre, written to this.gx / this.gz. */
    private gradAt(cx: number, cz: number): boolean {
      const f = this.front;
      const base = f[cx + (cz << SHIFT)];
      if (base >= UNREACHABLE) return false;
      const xp = this.at(cx + 1, cz, base), xm = this.at(cx - 1, cz, base);
      const zp = this.at(cx, cz + 1, base), zm = this.at(cx, cz - 1, base);
      // Cost rises away from the player, so descent is -(d cost).
      this.gx = (xm - xp) * 0.5;
      this.gz = (zm - zp) * 0.5;
      return true;
    }
  
    /** Neighbour cost; a wall reads as slightly worse than here, so the
     *  gradient pushes away from it instead of producing a NaN. */
    private at(cx: number, cz: number, base: number): number {
      const v = this.front[(cx & MASK) + ((cz & MASK) << SHIFT)];
      return v >= UNREACHABLE ? base + WALL_BIAS : v;
    }
  
    /* ---------------------------------------------------------------- HUD */
  
    stats() {
      return {
        ready: this.ready,
        building: this.building,
        openCells: this.open,
        lastNodes: this.lastNodes,
        lastBuildMs: this.lastBuildMs,
        dirtyColumns: this.dirtyList.length,
      };
    }
  }
  ```

  I read the whole doc, but scoped the work to Phase 5 only. Short version of the audit before the deliverable:

**Six real defects in Phase 5 as written**

1. `if (farthest) { …; return; }` runs whenever *any* hostile is alive — so after the first spawn the game never spawns again, it just teleports one enemy forever. Population freezes at 1.
2. `if (hostiles >= MAX_POP) return;` makes the recycle path dead code at the cap. The two mechanisms cancel each other out.
3. The cap is unreachable anyway: `wildTimer = 1.4 + rand*1.4`, one spawn per tick → 0.48/s → **87 minutes of continuous night** to reach 2,500. Raising `WILD_CAP` 12 → 2500 changes nothing without a rate ramp.
4. `MAX_POP_DISTANCE = 220` gates the *spawn ring* (24–48 m) — the check can never fail. It's dead code, and there's no despawn at any radius, so nothing bounds memory.
5. Recycling teleports the farthest enemy with no distance floor and no camp exclusion — it will yank a visible enemy across the screen, and it overwrites a camp member's `home`, permanently destroying its camp.
6. `const MAX_POP` is module-scoped, not exported; `engine.ts`'s `enemiesCap: MAX_POP` won't compile. Also, the rewritten `wildTick` silently drops the `gridStale = true` that Phase 3 required, and calls `this.attach(e)` instead of Phase 1's `attachToScene(sceneRef)`.
