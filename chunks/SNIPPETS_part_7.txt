```ts
  // ------------------------------------------------------------------- tree species
  /** Per-tree deterministic random in [0,1). */
  private j(wx: number, wz: number, salt: number): number {
    return hash2(this.seed ^ salt, wx, wz);
  }
  private jn(wx: number, wz: number, salt: number, i: number): number {
    return hash2(this.seed ^ salt, wx * 31 + i * 7, wz * 17 + i * 13);
  }
  private pickI(range: [number, number], t: number): number {
    return range[0] + Math.floor(t * Math.max(1, range[1] - range[0] + 1));
  }

  private placeTree(set: SetFn, wx: number, h: number, wz: number): void {
    const s = this.flora.tree;
    const tag = this.originTag;
    switch (s.silhouette) {
      case 'conifer':   return this.tConifer(set, wx, h, wz, s, tag);
      case 'palm':      return this.tPalm(set, wx, h, wz, s, tag);
      case 'spire':     return this.tSpire(set, wx, h, wz, s, tag);
      case 'crystal':   return this.tCrystal(set, wx, h, wz, s, tag);
      case 'succulent': return this.tSucculent(set, wx, h, wz, s, tag);
      case 'umbrella':  return this.tUmbrella(set, wx, h, wz, s, tag);
      case 'fungal':    return this.tFungal(set, wx, h, wz, s, tag);
      case 'none':      return;
      default:          return this.tBroadleaf(set, wx, h, wz, s, tag);
    }
  }

  /** Leaf ellipsoid with hash-thinned edges. Shared by most silhouettes. */
  private blob(
    set: SetFn, cx: number, cy: number, cz: number,
    rx: number, ry: number, rz: number, density: number, salt: number, tag: OriginTag,
  ): void {
    for (let dy = -ry; dy <= ry; dy++) {
      for (let dx = -rx; dx <= rx; dx++) {
        for (let dz = -rz; dz <= rz; dz++) {
          const d = (dx * dx) / Math.max(1, rx * rx)
                  + (dy * dy) / Math.max(1, ry * ry)
                  + (dz * dz) / Math.max(1, rz * rz);
          if (d > 1.05) continue;
          // Edge voxels thin out; core is solid → soft organic silhouette.
          const edge = d > 0.55;
          if (edge && hash2(this.seed ^ salt, cx * 31 + dx, (cy + dy) * 7 + dz) > density) continue;
          set(cx + dx, cy + dy, cz + dz, B.LEAVES, false, tag);
        }
      }
    }
  }

  /** Trunk with optional lean; returns the top voxel position. */
  private trunk(
    set: SetFn, wx: number, h: number, wz: number, len: number,
    lean: number, dirX: number, dirZ: number, tag: OriginTag,
  ): [number, number, number] {
    let tx = wx, tz = wz;
    for (let i = 1; i <= len; i++) {
      const t = i / Math.max(1, len);
      tx = wx + Math.round(dirX * lean * t * t);
      tz = wz + Math.round(dirZ * lean * t * t);
      set(tx, h + i, tz, B.LOG, true, tag);
    }
    return [tx, h + len, tz];
  }

  // --- broadleaf: the old placeOak, now parameterised ---------------------------
  private tBroadleaf(set: SetFn, wx: number, h: number, wz: number, s: TreeShape, tag: OriginTag) {
    const trunkH = this.pickI(s.trunkH, this.j(wx, wz, 0xa7ee));
    const [tx, top, tz] = this.trunk(set, wx, h, wz, trunkH, s.trunkLean,
      Math.cos(this.j(wx, wz, 0xa1) * 6.283), Math.sin(this.j(wx, wz, 0xa1) * 6.283), tag);
    const r = this.pickI(s.canopyR, this.j(wx, wz, 0xa2));
    this.blob(set, tx, top - 1, tz, r, Math.max(1, s.canopyH >> 1), r, s.leafDensity, 0x1eaf, tag);
    this.blob(set, tx, top + 1, tz, Math.max(1, r - 1), 1, Math.max(1, r - 1), s.leafDensity * 0.9, 0x1eb0, tag);
  }

  // --- conifer: the old placeSpruce, tapered cone --------------------------------
  private tConifer(set: SetFn, wx: number, h: number, wz: number, s: TreeShape, tag: OriginTag) {
    const trunkH = this.pickI(s.trunkH, this.j(wx, wz, 0xc0f1));
    this.trunk(set, wx, h, wz, trunkH, 0, 0, 0, tag);
    const rMax = s.canopyR[1];
    const base = h + Math.max(2, Math.round(trunkH * 0.32));
    for (let ly = base; ly <= h + trunkH + 1; ly++) {
      const t = (ly - base) / Math.max(1, h + trunkH + 1 - base);
      // Skirted cone: radius shrinks with height, with a 2-layer sawtooth for "branchiness".
      const rad = Math.max(0, Math.round(rMax * (1 - t) + ((ly & 1) ? 0 : 0.6)));
      if (rad <= 0) { set(wx, ly, wz, B.LEAVES, false, tag); continue; }
      for (let dx = -rad; dx <= rad; dx++)
        for (let dz = -rad; dz <= rad; dz++) {
          if (dx * dx + dz * dz > rad * rad + 1) continue;
          if (Math.abs(dx) + Math.abs(dz) >= rad * 2 &&
              hash2(this.seed ^ 0xc0f2, wx * 31 + dx, ly * 7 + dz) > s.leafDensity) continue;
          if (dx === 0 && dz === 0 && ly <= h + trunkH) continue;   // keep the trunk visible
          set(wx + dx, ly, wz + dz, B.LEAVES, false, tag);
        }
    }
  }

  // --- palm: bare curved trunk, radial drooping fronds (jungle / ocean) ---------
  private tPalm(set: SetFn, wx: number, h: number, wz: number, s: TreeShape, tag: OriginTag) {
    const trunkH = this.pickI(s.trunkH, this.j(wx, wz, 0x9a1));
    const a = this.j(wx, wz, 0x9a2) * Math.PI * 2;
    const [tx, top, tz] = this.trunk(set, wx, h, wz, trunkH, s.trunkLean,
      Math.cos(a), Math.sin(a), tag);
    const fronds = Math.max(4, s.branches);
    const len = this.pickI(s.canopyR, this.j(wx, wz, 0x9a3)) + 1;
    for (let f = 0; f < fronds; f++) {
      const fa = (f / fronds) * Math.PI * 2 + this.j(wx, wz, 0x9a4) * 0.9;
      const dx = Math.cos(fa), dz = Math.sin(fa);
      for (let i = 1; i <= len; i++) {
        // arc out then down — the palm signature
        const drop = Math.round(s.droop * (i * i) / Math.max(1, len));
        const px = tx + Math.round(dx * i), pz = tz + Math.round(dz * i);
        set(px, top - drop, pz, B.LEAVES, false, tag);
        if (i < len && this.jn(wx, wz, 0x9a5, f * 8 + i) < s.leafDensity) {
          // one-voxel side lobes so fronds read as feathered, not as sticks
          set(px + Math.round(-dz), top - drop, pz + Math.round(dx), B.LEAVES, false, tag);
        }
      }
    }
    set(tx, top + 1, tz, B.LEAVES, false, tag);
  }

  // --- spire: alien, tall spindly trunk + stacked small glowing blobs -----------
  private tSpire(set: SetFn, wx: number, h: number, wz: number, s: TreeShape, tag: OriginTag) {
    const trunkH = this.pickI(s.trunkH, this.j(wx, wz, 0x5p1 as unknown as number ?? 0x5a1));
    const a = this.j(wx, wz, 0x5a2) * Math.PI * 2;
    const [tx, top, tz] = this.trunk(set, wx, h, wz, trunkH, s.trunkLean,
      Math.cos(a), Math.sin(a), tag);
    const nodes = Math.max(2, s.branches);
    for (let n = 0; n < nodes; n++) {
      const t = 0.45 + (n / Math.max(1, nodes - 1)) * 0.55;
      const y = h + Math.round(trunkH * t);
