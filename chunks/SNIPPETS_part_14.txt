          item.vel.x += (dX * BELT_SPEED - item.vel.x) * k;
          item.vel.z += (dZ * BELT_SPEED - item.vel.z) * k;
          if (dX === 0) item.vel.x += (bx + 0.5 - p.x) * BELT_CENTERING * step;
          else          item.vel.z += (bz + 0.5 - p.z) * BELT_CENTERING * step;
        }
      }

      /* ---- horizontal collision: overlap-escape preserved exactly ---- */
      const stuck = this.blocked(p.x, p.y, p.z);

      const nx = p.x + item.vel.x * step;
      if (!stuck && this.blocked(nx, p.y, p.z)) item.vel.x = 0;
      else p.x = nx;

      const nz = p.z + item.vel.z * step;
      if (!stuck && this.blocked(p.x, p.y, nz)) item.vel.z = 0;
      else p.z = nz;

      if (!onBelt) {
        const drag = item.grounded ? 9 : 1.2;
        const f = Math.max(0, 1 - drag * step);
        item.vel.x *= f;
        item.vel.z *= f;
      }

      /* ---- visual transform: identical bob/spin ---- */
      item.spin += step * 2.4;
      const vy = p.y + 0.12 + Math.sin(item.time * 3 + item.spin) * 0.035;
      if (this.instancer) this.instancer.set(item, p.x, vy, p.z, item.spin);
      else if (item.mesh) {
        item.mesh.position.set(p.x, vy, p.z);
        item.mesh.rotation.y = item.spin;
      }

      /* ---- index membership: O(1), only when the cell actually changes ---- */
      this.grid.move(item);

      /* ---- S5b sleep test ---- */
      if (this.sleepEnabled && item.grounded && !onBelt && !stuck &&
          item.time > POP_TIME) {
        const v2 = item.vel.x * item.vel.x + item.vel.z * item.vel.z
                 + item.vel.y * item.vel.y;
        if (v2 < SLEEP_VEL2) {
          const dxp = minImageF(p.x - px);
          const dzp = minImageF(p.z - pz);
          if (dxp * dxp + dzp * dzp > SLEEP_DIST2) { this.sleep(item); continue; }
        }
      }
    }

    this.wakeSweep(px, pz);
    this.sweepPickup(playerPos);
    this.instancer?.flush();
  }

  /** Round-robin: a sleeper is rechecked at least every ~0.3 s at the 512 cap. */
  private wakeSweep(px: number, pz: number): void {
    const n = this.sleepers.length;
    if (n === 0) return;
    const checks = Math.min(WAKE_CHECKS_PER_FRAME, n);
    for (let c = 0; c < checks; c++) {
      if (this.wakeCursor >= this.sleepers.length) this.wakeCursor = 0;
      const it = this.sleepers[this.wakeCursor];
      this.stats.wakeChecks++;
      const dx = minImageF(it.pos.x - px);
      const dz = minImageF(it.pos.z - pz);
      if (dx * dx + dz * dz <= SLEEP_DIST2) { this.wake(it); continue; }
      this.wakeCursor++;
    }
  }

  /** Any voxel change may free, bury, or belt a settled item: wake its neighborhood. */
  private onBlockChanged(x: number, y: number, z: number): void {
    if (this.sleepers.length === 0) return;
    this.grid.forEachBox(x - 1.5, y - 1.5, z - 1.5,
                         x + 1.5, y + 1.5, z + 1.5,
                         this.wakeVisit);
  }
  private wakeVisit = (it: DroppedItem): void => { this.wake(it); };

  /* ------------------------------- player pickup ---------------------------- */

  private pickX = 0; private pickY = 0; private pickZ = 0;
  private pickBest: DroppedItem | null = null;
  private pickBestD2 = 0;

  private pickVisit = (it: DroppedItem): void => {
    this.stats.pickupCandidates++;
    if (it.time < POP_TIME) return;
    const dy = this.pickY - it.pos.y;
    if (dy > 1.6 || dy < -1.6) return;
    const dx = minImageF(this.pickX - it.pos.x);
    const dz = minImageF(this.pickZ - it.pos.z);
    const d2 = dx * dx + dz * dz;
    if (d2 > PICKUP_DIST * PICKUP_DIST) return;
    if (this.pickBest !== null && d2 >= this.pickBestD2) return;
    this.pickBest = it;
    this.pickBestD2 = d2;
  };

  /**
   * Was: a distance test for EVERY item, every frame, inside the physics loop.
   * Now: one bucket query around the player (~36 cells) per frame, nearest first.
   * Sleeping items are still in the grid, so a settled item is never unpickable.
   * The pickup predicates (POP_TIME gate, |dy| <= 1.6, min-image XZ, PICKUP_DIST)
   * are unchanged; only the visit order changes, from array order to nearest-first.
   */
  private sweepPickup(playerPos: THREE.Vector3): void {
    if (this.items.length === 0) return;
    this.stats.pickupQueries++;
    this.pickX = playerPos.x;
    this.pickY = playerPos.y + 0.9;
    this.pickZ = playerPos.z;

    // At most a handful of items can be picked in one frame; loop until the
    // inventory refuses or nothing is in range. Bounded by the cell contents.
    for (let guard = 0; guard < 8; guard++) {
      this.pickBest = null;
      this.pickBestD2 = Infinity;
      this.grid.forEachBox(
        this.pickX - PICKUP_DIST, this.pickY - 1.6, this.pickZ - PICKUP_DIST,
        this.pickX + PICKUP_DIST, this.pickY + 1.6, this.pickZ + PICKUP_DIST,
        this.pickVisit,
      );
      const it = this.pickBest;
      this.pickBest = null;
      if (it === null) return;
      if (!this.inv.addBlock(it.blockId, 1)) return;     // inventory full: stop
      this.audio.foley('snap');
      this.onPickup?.(it.blockId);
      this.despawn(it);
      this.stats.pickups++;
    }
  }
```

**Documented micro-delta.** The belt lookup now uses `peekBlock`, so an item resting on
*unloaded* voxel data is not belted until that chunk's data exists. Previously
`world.get()` would generate the chunk from inside item physics — a frame spike, at a
location the player cannot see (the item is outside the streamed radius by definition).
Streaming restores belt motion within a frame or two of the player approaching.

### 5.7 Optional: `src/game/fps/ItemInstancer.ts` (stage S5c, default off)

```ts
// src/game/fps/ItemInstancer.ts   (NEW FILE — opt-in via ItemDropOptions.instanced)
import * as THREE from 'three';
import type { DroppedItem } from './ItemDrop';

interface Group {
  mesh: THREE.InstancedMesh;
  owners: (DroppedItem | null)[];
  count: number;
  dirty: boolean;
}

const START_CAP = 64;

/**
 * One InstancedMesh per blockId geometry. Slots are dense: releasing slot s moves
 * the last live instance into s and fixes that item's `inst`, so `count` is always
 * the number of live instances and no gaps are ever drawn.
 *
 * Draw calls: O(distinct dropped block ids) instead of O(live items).
 * Tradeoff: instanced shadow casting draws every instance in the group, where
 * per-mesh items were frustum-culled individually. At the 512-item cap this is a
 * clear net win; it is behind a flag so it can be measured, not assumed.
 */
export class ItemInstancer {
  readonly stats = { groups: 0, instances: 0, grows: 0, writes: 0 };

  private groups = new Map<number, Group>();
  private m = new THREE.Matrix4();

  constructor(
    private readonly scene: THREE.Scene,
    private readonly geometryFor: (blockId: number) => THREE.BufferGeometry,
    private readonly materialFor: () => THREE.Material,
  ) {}

  private groupFor(blockId: number): Group {
    let g = this.groups.get(blockId);
    if (g) return g;
    g = this.make(blockId, START_CAP);
    this.groups.set(blockId, g);
    this.stats.groups = this.groups.size;
    return g;
  }

  private make(blockId: number, cap: number): Group {
    const mesh = new THREE.InstancedMesh(
      this.geometryFor(blockId), this.materialFor(), cap,
    );
    mesh.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
    mesh.castShadow = true;
    mesh.receiveShadow = false;
    mesh.frustumCulled = false;      // instances move independently of the origin
    mesh.count = 0;
    this.scene.add(mesh);
    return { mesh, owners: new Array(cap).fill(null), count: 0, dirty: false };
