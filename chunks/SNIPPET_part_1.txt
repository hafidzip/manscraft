## 1. Engine Main Loop (`src/game/engine.ts` — `tick()`)

**What changed**

- One **wall-clock deadline** for the whole simulation half of the frame (`simDeadline`), so a slow system steals from itself rather than from the next frame.
- Optional systems (fluid, particles, sky, snapshot) **carry their skipped `dt`** instead of dropping it — flow/spawn/animation rates stay identical over time, only granularity changes.
- Streaming budget now derives from **time actually left** rather than fps buckets (fps is a lagging EMA; by the time it drops the jank already happened).
- Post pipeline degrades in **two tiers with hysteresis + a dwell timer**, and the output stage is re-routed to the last live RT so skipping a pass is never a stale-frame artifact.
- Deliberately **not** alternating GI/volumetric on odd/even frames: `lumenLite.renderPipeline` *is* the lighting composite (freezing it ghosts the whole screen), and toggling a full-screen light-shaft pass at 30 Hz strobes. Hysteresis + dwell gives the same average saving without the flicker. Hook for a temporal 2-frame volumetric stride is left in place if you later add reprojection.
- Gameplay-critical systems (`tickPlay`, `enemies`, `flow`, `world`) are never shed.

```ts
// ─── module scope (top of engine.ts) ────────────────────────────────────────
const FRAME_TARGET_MS  = 16.6;  // 60 fps
const SIM_SOFT_MS      = 8.0;   // sim above this -> shed volumetric
const SIM_HARD_MS      = 11.0;  // sim above this -> shed volumetric + bloom
const QUALITY_DWELL_MS = 250;   // min ms between quality switches (anti-strobe)
const SNAPSHOT_MIN_MS  = 6.0;   // headroom required before a snapshot readback
const Q_FULL = 0, Q_NO_VOL = 1, Q_MINIMAL = 2;

// ─── new engine fields ──────────────────────────────────────────────────────
private simMsEma = 0;                    // EMA of simulation cost, ms
private postQ: 0 | 1 | 2 = Q_FULL;
private postQAt = 0;
private dtFluid = 0;                     // carried dt for shed-able systems
private dtParticles = 0;
private dtSky = 0;
private dtBloom = 0;
private dtSnap = 0;

// ─── tick ───────────────────────────────────────────────────────────────────
private tick = (): void => {
    if (this.disposed) return;

    const frameStart = performance.now();
    const dt = Math.min(this.clock.getDelta(), 0.05);
    const fpsNow = 1 / Math.max(dt, 1e-4);
    this.fps += (fpsNow - this.fps) * 0.06;
    this.time += dt;

    // Single simulation deadline for the frame. Cheap to read, checked only at
    // system boundaries (never per-item).
    const simDeadline = frameStart + SIM_SOFT_MS;

    if (this.locked) this.tickPlay(dt);
    else this.tickMenuCamera(dt);
    if (this.dmgFollowT > 0) this.dmgFollowT = Math.max(0, this.dmgFollowT - dt);

    // ... space exit check (unchanged) ...

    WATER_TIME.value += dt;
    if (this.textures) {
      this.textures.water.offset.x += dt * 0.003;
      this.textures.water.offset.y += dt * 0.006;
      animateConveyorTiles(this.textures, this.time);
    }

    // --- Fluid: deadline-gated, dt carried so flow speed is unchanged --------
    this.dtFluid += dt;
    if (this.locked && performance.now() < simDeadline) {
      this.fluid.update(this.dtFluid);
      this.dtFluid = 0;
    }

    if (this.locked) {
      let streamX = this.player.pos.x;
      let streamZ = this.player.pos.z;
      if (this.piloting && this.ship) {
        streamX = this.ship.pos.x + this.ship.vel.x * 0.7;
        streamZ = this.ship.pos.z + this.ship.vel.z * 0.7;
      }

      // Streaming gets the time that is actually left, clamped to the old range.
      const left = simDeadline - performance.now();
      const frameBudget = Math.max(1.0, Math.min(5.5, left));
      this.world.update(streamX, streamZ, frameBudget);
      this.world.syncChunkOffsets(this.camera.position.x, this.camera.position.z);

      // Flow field (logic unchanged — gameplay critical)
      const fpx = this.player.pos.x, fpy = this.player.pos.y, fpz = this.player.pos.z;
      const mdx = C.wrapDelta(fpx - this.lastFieldX, C.WORLD_SIZE);
      const mdz = C.wrapDelta(fpz - this.lastFieldZ, C.WORLD_SIZE);
      const fieldMoved = !(mdx * mdx + mdz * mdz < 64)
                      || !(Math.abs(fpy - this.lastFieldY) < 4);
      if (fieldMoved) {
        this.flow.invalidate();
        this.lastFieldX = fpx; this.lastFieldY = fpy; this.lastFieldZ = fpz;
      }
      this.flow.update(fpx, fpy, fpz, this.world);
      if (this.ship && !this.piloting) this.ship.updateParked(dt);
    }

    // Enemies: never shed (gameplay visible).
    if (this.locked && (!this.piloting || this.shipAltitude() < 26)) {
      this.enemies.update(dt);
    }

    if (this.locked) {
      this.updateFurnaces(dt);
      if (this.openFurnaceKey) this.events.onStats(this.buildStats());
      this.updateMerchantProximity();
      this.updateShop();

      // Cosmetic systems: shed past the deadline, dt carried.
      const cosmeticsOk = performance.now() < simDeadline;
      this.dtParticles += dt;
      this.dtSky += dt;
      if (cosmeticsOk) {
        this.particles.update(this.dtParticles); this.dtParticles = 0;
        this.sky.update(this.dtSky, this.camera.position); this.dtSky = 0;
      } else if (this.dtParticles > 0.1) {
        // Hard floor: never let particles/sky stall for more than ~6 frames.
        this.particles.update(this.dtParticles); this.dtParticles = 0;
        this.sky.update(this.dtSky, this.camera.position); this.dtSky = 0;
      }
    }

    // ... shadow updates, fog uniforms, grass time, volumetric, bloom uniforms ...
    // (uniform writes are ~free; leave them unconditional so a quality change
    //  never shows a one-frame stale uniform)

    // --- Decide post-processing tier for THIS frame -------------------------
    const simMs = performance.now() - frameStart;
    this.simMsEma += (simMs - this.simMsEma) * 0.15;
    this.updatePostQuality(frameStart);

    // --- RENDER PIPELINE: passes are shed from the tail, output re-routed ---
    if (this.mainRT) {
      const q = this.postQ;

      this.renderer.setRenderTarget(this.mainRT);
      this.renderer.render(this.scene, this.camera);

      // 1) Lighting + GI. Never skipped: this is the lighting composite, and a
      //    stale lightingRT ghosts the entire screen.
      this.lumenLite!.renderPipeline(
        this.renderer, this.mainRT, this.lightingRT!, this.giRT!, this.blurRT!);

      // 2) Depth fog: one cheap pass, drives readability. Never skipped.
      this.depthFogPass!.render(
        this.renderer, this.fogRT!, this.lightingRT!, this.mainRT.depthTexture);

      let finalTex: THREE.Texture = this.fogRT!.texture;

      // 3) Bloom (in-place on fogRT). dt is accumulated so its temporal
      //    response is unchanged across skipped frames.
      this.dtBloom += dt;
      if (q < Q_MINIMAL) {
        this.bloom!.render(this.renderer, this.fogRT!, this.fogRT!, this.dtBloom, false);
        this.dtBloom = 0;
      }

      // 4) Volumetric light -> volumetricRT.
      if (q === Q_FULL) {
        this.volumetricLight!.render(this.renderer, this.volumetricRT!, this.fogRT!);
        finalTex = this.volumetricRT!.texture;
      }

      // 5) Output stage always runs, reading the last RT still in the chain.
      this.outputStage!.render(this.renderer, finalTex);
    }

    // Snapshot readback can force a GPU sync — only with real headroom left.
    this.dtSnap += dt;
    if (performance.now() - frameStart < FRAME_TARGET_MS - SNAPSHOT_MIN_MS) {
      this.captureSnapshot(this.dtSnap);
      this.dtSnap = 0;
    }
};
