**Capacity proof.** Phase 2 already caps the tiers by count, and LOD is granted by tier, so the instance count is bounded by construction:

```
96 hot   × 6 parts =   576
400 warm × 3 parts = 1,200
2,004 cold × 1 part = 2,004
                      -----
                      3,780  worst case at 2,500 rendered
```

`MAX_INSTANCES = 6144` gives 60% headroom and stays a power of two. Buffers: 393 KB matrices + 74 KB colours. This coupling is the reason phase 2 is a hard prerequisite — without the count cap, a crowd that all closes to 24 m demands 15,000 instances.

### 6.2 — The instancer

```ts
// src/game/fps/EnemyInstancer.ts
import * as THREE from 'three';
import { LOD_PARTS, PALETTE, PRESET_SCALE } from './EnemyRig';

export const MAX_INSTANCES = 6144;   // see the capacity proof in 6.1

export interface InstancedAgent {
  presetIdx: number;
  lod: 0 | 1 | 2;
  yaw: number;        // radians
  gait: number;       // radians, advanced by the existing walk animation
  gaitGain: number;   // 0..1, scales with horizontal speed
  hitFlash: number;   // seconds remaining
}

/** r159+ uses addUpdateRange(); older builds use updateRange. Units are ARRAY
 *  ELEMENTS, not bytes and not instances. */
function markRange(attr: THREE.BufferAttribute, count: number) {
  const a = attr as any;
  if (typeof a.addUpdateRange === 'function') {
    a.clearUpdateRanges();
    a.addUpdateRange(0, count);
  } else if (a.updateRange) {
    a.updateRange.offset = 0;
    a.updateRange.count = count;
  }
  attr.needsUpdate = true;
}

export class EnemyInstancer {
  readonly mesh: THREE.InstancedMesh;
  private readonly mat: Float32Array;
  private readonly col: Float32Array;
  private live = 0;
  private dropped = 0;

  constructor(material?: THREE.Material) {
    const geo = new THREE.BoxGeometry(1, 1, 1);              // 24 verts, 12 tris
    const m = material ?? new THREE.MeshLambertMaterial({ color: 0xffffff });
    this.mesh = new THREE.InstancedMesh(geo, m, MAX_INSTANCES);
    this.mesh.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
    this.mesh.instanceColor = new THREE.InstancedBufferAttribute(
      new Float32Array(MAX_INSTANCES * 3).fill(1), 3,
    );
    this.mesh.instanceColor.setUsage(THREE.DynamicDrawUsage);

    this.mesh.count = 0;
    this.mesh.frustumCulled = false;      // bug #2
    this.mesh.matrixAutoUpdate = false;   // the container never moves
    this.mesh.updateMatrix();
    this.mesh.castShadow = true;
    this.mesh.receiveShadow = true;
    this.mesh.raycast = () => {};         // bug #6 — hit-testing goes via phase 3's grid

    this.mat = this.mesh.instanceMatrix.array as Float32Array;
    this.col = this.mesh.instanceColor.array as Float32Array;
  }

  get liveCount() { return this.live; }
  get droppedCount() { return this.dropped; }

  begin() { this.live = 0; this.dropped = 0; }

  /**
   * Append one agent's parts. (rx, ry, rz) is the MINIMUM-IMAGE render position
   * (bug #4), not the raw world position. Returns false when the buffer is full.
   *
   * There is no slot map and no swap-remove: the whole prefix is rebuilt every
   * frame. At ~30 ns per instance that is cheaper than dirty-tracking, and it
   * makes death, spawning, phase 5 recycling, tier flips and LOD flips free —
   * an entire class of stale-slot bugs simply cannot occur.
   */
  push(e: InstancedAgent, rx: number, ry: number, rz: number): boolean {
    const rig = LOD_PARTS[e.lod];
    const need = rig.length;
    if (this.live + need > MAX_INSTANCES) { this.dropped++; return false; }   // bug #3

    const k = PRESET_SCALE[e.presetIdx];
    const yc = Math.cos(e.yaw), ys = Math.sin(e.yaw);

    // Two sin/cos pairs at LOD0, one at LOD1, none at LOD2 — trig cost falls
    // off with distance exactly like everything else.
    let c1 = 1, s1 = 0, c2 = 1, s2 = 0;
    if (e.lod < 2) {
      const amp = (e.lod === 0 ? 0.90 : 0.35) * e.gaitGain;
      const phi = Math.sin(e.gait) * amp;
      c1 = Math.cos(phi); s1 = Math.sin(phi);
      if (e.lod === 0) { const a = phi * 0.7; c2 = Math.cos(a); s2 = Math.sin(a); }
    }

    const flash = e.hitFlash > 0 ? Math.min(1, e.hitFlash * 5) : 0;
    const pbase = e.presetIdx * 9;
    const m = this.mat, c = this.col;
    let o = this.live * 16, co = this.live * 3;

    for (let i = 0; i < need; i++) {
      const p = rig[i];

      let cf = 1, sf = 0;
      const sw = p.swing;
      if (sw ===  1) { cf = c1; sf =  s1; }
      else if (sw === -1) { cf = c1; sf = -s1; }
      else if (sw ===  2) { cf = c2; sf =  s2; }
      else if (sw === -2) { cf = c2; sf = -s2; }

      const bx = p.sx * k, by = p.sy * k, bz = p.sz * k;
      const ax = p.ax * k, ay = p.ay * k, az = p.az * k;
      const off = p.hang ? by * 0.5 : 0;

      // M = T(render) · Ry(yaw) · T(pivot) · Rx(swing) · T(0,-off,0) · S(box)
      // written straight into the buffer, column-major, no Matrix4 churn.
