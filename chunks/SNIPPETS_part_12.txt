  /** Active-planet path: full-rate, in lockstep with the engine frame. */
  tickLive(dt: number): void {
    if (this.backlogSec > 0) this.advance(dt, 6);   // absorb absence first, budgeted
    else this.step(Math.min(dt, MAX_STEP));
  }

  /* ---- live attachment ------------------------------------------------- */
  attachToLive(world: LiveWorldLike, engineFurnaces: Map<string, FurnaceState>): void {
    this.live = world;
    this.view.live = world;
    this.view.invalidateAll();
    this.rebuildFromDeltas();
    // Sim state wins; engine map becomes a mirror the UI reads.
    engineFurnaces.clear();
    for (const [k, st] of this.furnaces) engineFurnaces.set(k, st);
    this.syncFurnaceBlocks();
  }

  /** Adopt anything the live engine created while attached, then go headless. */
  mergeFromLive(engineFurnaces: Map<string, FurnaceState>): void {
    for (const [k, st] of engineFurnaces) if (!this.furnaces.has(k)) this.furnaces.set(k, st);
    this.rebuildFromDeltas();
    this.live = null;
    this.view.live = null;
    this.view.invalidateAll();
  }

  private syncFurnaceBlocks(): void {
    if (!this.live) return;
    for (const [k, st] of this.furnaces) {
      const p = k.split(','); const x = +p[0], y = +p[1], z = +p[2];
      const cur = this.live.peekBlock(x, y, z);
      if (cur !== B.FURNACE && cur !== B.FURNACE_LIT) continue;
      const want = st.burn > 0 ? B.FURNACE_LIT : B.FURNACE;
      if (cur !== want) this.live.setBlock(x, y, z, want);
    }
  }

  /* ---- views for the live agents (visuals become a VIEW, not truth) ----- */
  inserterView(key: number): { state: number; t: number; held: number } | null {
    const m = this.machines.get(key);
    return m && (m.kind & MK_INSERTER) ? { state: m.st, t: m.t, held: m.held } : null;
  }
  minerView(key: number): { charge: number; tx: number; ty: number; tz: number } | null {
    const m = this.machines.get(key);
    return m && (m.kind & MK_LASER) ? { charge: m.charge, tx: m.tx, ty: m.ty, tz: m.tz } : null;
  }

  /* ---- persistence ------------------------------------------------------ */
  snapshot(): SimSnapshot {
    const flat: number[] = [];
    for (const m of this.machines.values()) flat.push(m.key, m.st, m.t, m.held, m.charge, m.tx, m.ty, m.tz);
    return {
      v: 1, simTimeSec: this.simTimeSec, backlogSec: this.backlogSec, lastWallMs: Date.now(),
      machines: flat, produced: this.stats.produced, mined: this.stats.mined,
    };
  }

  restore(snap: SimSnapshot | null, furnaces: Map<string, FurnaceState>): void {
    this.furnaces.clear();
    for (const [k, st] of furnaces) this.furnaces.set(k, st);
    this.rebuildFromDeltas();
    if (!snap || snap.v !== 1) return;
    this.simTimeSec = snap.simTimeSec;
    this.backlogSec = snap.backlogSec;
    this.lastWallMs = snap.lastWallMs;
    this.stats.produced = snap.produced; this.stats.mined = snap.mined;
    for (let i = 0; i + 7 < snap.machines.length; i += 8) {
      const m = this.machines.get(snap.machines[i]);
      if (!m) continue;
      m.st = snap.machines[i + 1]; m.t = snap.machines[i + 2]; m.held = snap.machines[i + 3];
      m.charge = snap.machines[i + 4]; m.tx = snap.machines[i + 5]; m.ty = snap.machines[i + 6]; m.tz = snap.machines[i + 7];
    }
  }
}

## 5. `src/game/factory/universeSim.ts` — the global clock

```ts
/**
 * src/game/factory/universeSim.ts
 *
 * UniverseSim — module-scope singleton (like `session`) that outlives GameEngine/GameCanvas
 * remounts, so the factory keeps producing while the player is in space or on another planet.
 *
 * Clock policy:
 *   • wall-clock deltas from Date.now() — background-tab throttling of setInterval is
 *     therefore harmless (time is measured, not counted in ticks).
 *   • the ACTIVE planet ticks from the GameEngine frame (tickLive), full rate.
 *   • inactive planets accrue backlog and burn it round-robin under an 8 ms/tick budget.
 *   • cross-session absence is credited on hydrate(), capped at MAX_OFFLINE_SEC.
 *
 * OUT OF SCOPE: fluids and enemies remain live-only (see factorySim.ts header).
 */
import { PlanetFactorySim, type SimContext } from './factorySim';
import { ItemLedger } from './itemLedger';
import { WorldDeltaStore } from '../persist/worldDelta';
import { hub, themeFromJson, type PlanetSave } from '../persist/planetStore';
import { TerrainGenerator, planetSeedToWorldSeed } from '../world/generator';
import { deserializeFurnaces } from '../persist/worldDelta';

const BG_INTERVAL_MS = 250;
const BG_BUDGET_MS = 8;
const MAX_GAP_SEC = 60 * 60 * 8;      // cap a single wall-clock gap (sleep/suspend)
export const MAX_OFFLINE_SEC = 60 * 60 * 8;

export class UniverseSim {
  private sims = new Map<string, PlanetFactorySim>();
  private rr: string[] = [];
  private rrCursor = 0;
  private activeKey: string | null = null;
  private lastWallMs = Date.now();
  private timer: ReturnType<typeof setInterval> | null = null;
  readonly stats = { planets: 0, bgTicks: 0, bgSimSec: 0, activeSec: 0, saves: 0 };

  start(): void {
    if (this.timer !== null || typeof window === 'undefined') return;
    this.lastWallMs = Date.now();
    this.timer = setInterval(() => this.tickBackground(), BG_INTERVAL_MS);
  }
  stop(): void { if (this.timer !== null) { clearInterval(this.timer); this.timer = null; } }

  get(key: string): PlanetFactorySim | null { return this.sims.get(key) ?? null; }
  keys(): string[] { return Array.from(this.sims.keys()); }

  /**
   * Create (or fetch) the sim for a planet. `deltas`/`ledger` are the SAME instances the live
   * World and ItemDropManager use — one source of truth, no copying on transitions.
   */
  ensure(key: string, opts: {
    seed: number; themeJson: string | null;
    deltas?: WorldDeltaStore; ledger?: ItemLedger; save?: PlanetSave | null;
  }): PlanetFactorySim {
    const existing = this.sims.get(key);
    if (existing) return existing;
    const save = opts.save ?? hub.get(key);
    const deltas = opts.deltas ?? WorldDeltaStore.deserialize(save?.deltas);
    const ledger = opts.ledger ?? ItemLedger.deserialize(save?.ledger);
    const themeJson = opts.themeJson ?? save?.themeJson ?? null;
    const ctx: SimContext = {
      planetKey: key, seed: opts.seed, deltas, ledger,
      makeGenerator: () => {
        const theme = themeFromJson(themeJson);
        const seed = theme ? planetSeedToWorldSeed(theme.seed) : opts.seed;
        return new TerrainGenerator(seed, theme);
      },
    };
    const sim = new PlanetFactorySim(ctx);
    sim.restore(save?.sim ?? null, deserializeFurnaces(save?.furnaces ?? null));
    // Credit offline time (bounded) so a returning player sees real production.
    if (save?.savedAtMs) sim.enqueue(Math.min((Date.now() - save.savedAtMs) / 1000, MAX_OFFLINE_SEC));
    this.sims.set(key, sim);
    this.rr = Array.from(this.sims.keys());
    this.stats.planets = this.sims.size;
    return sim;
  }

  setActive(key: string | null): void {
    this.activeKey = key;
    this.lastWallMs = Date.now();
  }

  /** Called from the GameEngine main loop, next to this.machines.update(dt, ...). */
  tickActive(dt: number): void {
    if (!this.activeKey) return;
    const sim = this.sims.get(this.activeKey);
    if (!sim) return;
    sim.tickLive(dt);
    this.stats.activeSec += dt;
  }

  private tickBackground(): void {
    const nowMs = Date.now();
    const wallDt = Math.min((nowMs - this.lastWallMs) / 1000, MAX_GAP_SEC);
    this.lastWallMs = nowMs;
    if (wallDt <= 0) return;
    this.stats.bgTicks++;

    for (const [key, sim] of this.sims) if (key !== this.activeKey) sim.enqueue(wallDt);

    if (this.rr.length !== this.sims.size) this.rr = Array.from(this.sims.keys());
    const t0 = performance.now();
    let visited = 0;
    while (visited < this.rr.length && performance.now() - t0 < BG_BUDGET_MS) {
      const key = this.rr[this.rrCursor++ % this.rr.length];
      visited++;
      if (key === this.activeKey) continue;
      const sim = this.sims.get(key);
      if (!sim || sim.backlogSec <= 0) continue;
      this.stats.bgSimSec += sim.advance(0, Math.max(1, BG_BUDGET_MS / 2));
    }

    // Opportunistic delta self-shrink + periodic durability flush (cheap, off the hot path).
    if (this.stats.bgTicks % 40 === 0) {
      for (const [key, sim] of this.sims) { if (key !== this.activeKey) { hub.captureSim(key, sim); this.stats.saves++; } }
      void hub.flushNow();
    }
