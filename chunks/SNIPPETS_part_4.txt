}

// ------------------------------------------------------------------------ helpers
function mixHex(a: number, b: number, t: number): number {
  const ch = (h: number, s: number) => (h >> s) & 0xff;
  const m = (s: number) => Math.round(ch(a, s) * (1 - t) + ch(b, s) * t) & 0xff;
  return (m(16) << 16) | (m(8) << 8) | m(0);
}
