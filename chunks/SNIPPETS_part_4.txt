/** Rotation that looks from `dir` back at the origin — the light's basis. */
function lightQuat(dir: THREE.Vector3): THREE.Quaternion {
  _m.lookAt(_eye.copy(dir), _zero.set(0, 0, 0), Math.abs(dir.y) > 0.99 ? _altUp : _up);
  return _q.setFromRotationMatrix(_m);
}

/**
 * Centre an ortho shadow frustum on `centre`, quantised to WHOLE shadow texels in
 * the light's own basis. With a constant `dir` (Task 2) the shadow matrix can then
 * only ever change by an integer number of texels, which is the definition of a
 * swim-free shadow. `outCell` receives the integer texel cell so callers can detect
 * "the frustum actually moved" and skip a re-render otherwise.
 */
export function snapShadowCamera(
  cam: THREE.Camera & { left: number; right: number; top: number; bottom: number;
                        updateProjectionMatrix(): void },
  dir: THREE.Vector3, centre: THREE.Vector3,
  halfExtent: number, mapSize: number, distance: number,
  outCell?: THREE.Vector2,
): void {
  const q = lightQuat(dir);
  _qi.copy(q).invert();
  const texel = (halfExtent * 2) / mapSize;
  _v.copy(centre).applyQuaternion(_qi);
  const cx = Math.round(_v.x / texel);
  const cy = Math.round(_v.y / texel);
  if (outCell) outCell.set(cx, cy);
  _v.set(cx * texel, cy * texel, _v.z).applyQuaternion(q);

  cam.position.copy(_v).addScaledVector(dir, distance);
  cam.quaternion.copy(q);
  cam.left = -halfExtent; cam.right = halfExtent;
  cam.top = halfExtent;   cam.bottom = -halfExtent;
  cam.updateProjectionMatrix();
  cam.updateMatrixWorld(true);
}

/** Same snap, applied to a DirectionalLight's position/target pair. */
export function snapShadowLight(
  light: THREE.DirectionalLight, dir: THREE.Vector3, centre: THREE.Vector3,
  halfExtent: number, mapSize: number, distance: number, outCell?: THREE.Vector2,
): void {
  const q = lightQuat(dir);
  _qi.copy(q).invert();
  const texel = (halfExtent * 2) / mapSize;
  _v.copy(centre).applyQuaternion(_qi);
  const cx = Math.round(_v.x / texel);
  const cy = Math.round(_v.y / texel);
  if (outCell) outCell.set(cx, cy);
  _v.set(cx * texel, cy * texel, _v.z).applyQuaternion(q);

  light.target.position.copy(_v);
  light.position.copy(_v).addScaledVector(dir, distance);
  light.target.updateMatrixWorld();
  light.updateMatrixWorld();
}

import * as THREE from 'three';
import {
  CELESTIAL_SHADOW_SIZE, CELESTIAL_SHADOW_HALF_EXTENT, CELESTIAL_SHADOW_DISTANCE,
  SHADOW_FAR_SIZE, SHADOW_FAR_HALF_EXTENT, SHADOW_FAR_DISTANCE, SHADOW_FAR_RECENTER,
  SHADOW_CASCADE_BLEND, SHADOW_FAR_FADE, LAYER_FAR_CASTER,
} from '../engine/constants';
import { snapShadowCamera, snapShadowLight } from './shadowSnap';

/** Shared with every shadow-receiving material via `patchReceiver`. */
export const SHADOW_UNIFORMS = {
  uFarShadowMap:        { value: null as THREE.Texture | null },
  uFarShadowMatrix:     { value: new THREE.Matrix4() },
  uFarShadowTexel:      { value: new THREE.Vector2(1 / SHADOW_FAR_SIZE, 1 / SHADOW_FAR_SIZE) },
  uFarShadowBias:       { value: -0.0018 },
  uFarShadowNormalBias: { value: 0.55 },                                  // ~3 far texels
  uCascadeBlend:        { value: new THREE.Vector2(...SHADOW_CASCADE_BLEND) },
  uFarShadowFade:       { value: new THREE.Vector2(...SHADOW_FAR_FADE) },
  uFarShadowStrength:   { value: 1 },
};

/**
 * Injects the far cascade into any lit material. Runs AFTER
 * <lights_fragment_begin> (all direct light is accumulated, indirect is not yet)
 * so exactly the direct term is attenuated — material-type agnostic.
 */
export function patchReceiver(shader: THREE.WebGLProgramParametersWithUniforms): void {
  Object.assign(shader.uniforms, SHADOW_UNIFORMS);

  shader.vertexShader = shader.vertexShader
    .replace('#include <common>', `#include <common>
      uniform mat4 uFarShadowMatrix;
      uniform float uFarShadowNormalBias;
      varying vec4 vFarShadowCoord;
      varying float vFarViewDist;`)
    .replace('#include <project_vertex>', `#include <project_vertex>
      {
        vec4 fsWp = modelMatrix * vec4( transformed, 1.0 );
        vec3 fsN  = normalize( mat3( modelMatrix ) * objectNormal );
        vFarShadowCoord = uFarShadowMatrix * vec4( fsWp.xyz + fsN * uFarShadowNormalBias, 1.0 );
        vFarViewDist = -mvPosition.z;   // linear view depth, standard CSM split axis
      }`);

  shader.fragmentShader = shader.fragmentShader
    .replace('#include <packing>', `#include <packing>
      uniform sampler2D uFarShadowMap;
      uniform vec2 uFarShadowTexel;
      uniform float uFarShadowBias;
      uniform vec2 uCascadeBlend;
      uniform vec2 uFarShadowFade;
      uniform float uFarShadowStrength;
      varying vec4 vFarShadowCoord;
      varying float vFarViewDist;

      float farCascadeShadow() {
        vec3 sc = vFarShadowCoord.xyz / vFarShadowCoord.w;
        if ( sc.z > 1.0 || sc.z < 0.0 ) return 1.0;
        if ( any( lessThan( sc.xy, vec2( 0.0 ) ) ) || any( greaterThan( sc.xy, vec2( 1.0 ) ) ) ) return 1.0;
        float d = sc.z + uFarShadowBias;
        vec2 t = uFarShadowTexel;
        float s = 0.0;
        s += step( d, unpackRGBAToDepth( texture2D( uFarShadowMap, sc.xy + vec2( -0.5, -0.5 ) * t ) ) );
        s += step( d, unpackRGBAToDepth( texture2D( uFarShadowMap, sc.xy + vec2(  0.5, -0.5 ) * t ) ) );
        s += step( d, unpackRGBAToDepth( texture2D( uFarShadowMap, sc.xy + vec2( -0.5,  0.5 ) * t ) ) );
        s += step( d, unpackRGBAToDepth( texture2D( uFarShadowMap, sc.xy + vec2(  0.5,  0.5 ) * t ) ) );
        return s * 0.25;
      }`)
    .replace('#include <lights_fragment_begin>', `#include <lights_fragment_begin>
      {
        // 0 in the near field (three's own cascade owns it), 1 in the mid field,
        // back to 0 past uFarShadowFade.y -> no hard cascade seam, no hard end.
        float fw = smoothstep( uCascadeBlend.x, uCascadeBlend.y, vFarViewDist )
                 * ( 1.0 - smoothstep( uFarShadowFade.x, uFarShadowFade.y, vFarViewDist ) );
        if ( fw > 0.001 ) {
          float fs = mix( 1.0, farCascadeShadow(), fw * uFarShadowStrength );
          reflectedLight.directDiffuse  *= fs;
          reflectedLight.directSpecular *= fs;
        }
      }`);
}
