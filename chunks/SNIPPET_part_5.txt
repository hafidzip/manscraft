      if (e.alive && acc < period[e.tier]) { e.tickAccum = acc; continue; }
      e.tickAccum = 0;

      if (e.tier === TIER_DORMANT) { this.coarseAdvance(e, acc, ppx, ppz); continue; }

      // Fast path: at 60 fps a hot enemy always lands here (acc ~= 1/60 < 1/15),
      // so the divide + ceil + Math.min chain disappears for the common case.
      let n: number, sdt: number;
      if (acc <= MAX_TICK_DT) { n = 1; sdt = acc; }
      else {
        n = Math.min(MAX_SUBSTEPS, Math.ceil(acc / MAX_TICK_DT));   // exact old math
        sdt = Math.min(acc / n, MAX_TICK_DT);
      }

      let keep = true;
      for (let s = 0; s < n && keep; s++) keep = e.update(sdt, player);

      if (!keep) {
        const g = e.group;
        g.parent?.remove(g);
        const last = list.length - 1;     // swap-remove: O(1) instead of O(N) splice
        if (i !== last) list[i] = list[last];
        list.pop();
        if (!e.cfg.peaceful) this._hostiles = this._hostiles > 0 ? this._hostiles - 1 : 0;
      } else if (e.alive) {
        active.push(e);
      }
    }

    this.grid.build(list, ppx, ppz);
    this.separateGrid(active, dt);

    this.respawnTick(dt);
    this.wildTick(dt);

    // Camp proximity — zero allocations and zero sqrt for out-of-range camps.
    const camps = this.camps;
    const py = player.pos.y;
    for (let ci = 0; ci < camps.length; ci++) {
      const camp = camps[ci];
      const squad = camp.squad;
      if (camp.cleared || squad.length === 0) continue;

      const site = camp.site;
      const cdx = wrapDelta(ppx - site.cx, WORLD_SIZE);
      const cdz = wrapDelta(ppz - site.cz, WORLD_SIZE);
      const r = site.radius;
      if (cdx * cdx + cdz * cdz > r * r) continue;   // was Math.hypot(...) <= r

      CAMP_IMG.set(site.cx + cdx, py, site.cz + cdz);
      for (let si = 0; si < squad.length; si++) {
        const s = squad[si];
        // If investigate() copies into its own state (target.copy(p)) rather than
        // retaining the reference, drop the .clone() and pass CAMP_IMG directly —
        // that removes the last per-frame allocation in this method.
        if (s.alive) s.investigate(CAMP_IMG.clone());
      }
    }

    if (INSTANCED_ENEMIES) this.syncInstances(ppx, ppz);
}
```

**What changed**

- **`splice(i, 1)` → swap-remove.** O(N) memmove per death becomes O(1). With a squad wipe of 20 enemies at 96 population that's ~1900 element moves saved per frame-of-death.
- **Camp `Vector3` allocation hoisted** to module scope, and the range test uses squared distance — no `Math.hypot` (a libm call with overflow guarding) and no `sqrt` per camp per frame. Camps out of range now cost 2 multiplies and a compare.
- **`for...of` → indexed loops** on `camps` and `camp.squad`: no iterator objects, and `squad.length`/`camps.length` stay in registers.
- **Substep fast path** avoids `ceil`, a divide and two `Math.min` calls for every hot enemy at normal framerates, while keeping the original math bit-for-bit on long frames.
- **`tickAccum` written once** per enemy (was read + written on the skip path too), and `this.player` / `this.enemies` / `TIER_PERIOD_S` hoisted out of the loop.
- `_hostiles` clamp uses a compare instead of `Math.max`.

**On fusing `assignTiers` into the tick loop:** don't — not without a behaviour change. Tier assignment is budget-capped (`HOT_BUDGET`, `WARM_BUDGET`, `COLD_BUDGET`), which means it needs every enemy's distance *before* it can decide any enemy's tier. The cheap win instead: have `assignTiers` stash the squared player distance it already computes into a reusable `Float32Array` field (`this.distSq`, grown on population change), then feed that to `grid.build` and `coarseAdvance` instead of recomputing `wrapDelta` + distance a second and third time. That removes two O(N) distance passes without touching the tier semantics.

## 6. EnemyInstancer Sync (`src/game/fps/EnemyInstancer.ts` — `push`)

```ts
// ─── one-time init requirement (in the constructor / mesh allocation) ───────
// Direct array writes cannot lazily create instanceColor the way setColorAt does,
// so allocate it up front:
//   this.mesh.instanceColor ??= new THREE.InstancedBufferAttribute(
//     new Float32Array(maxInstances * 3).fill(1), 3);
//
// Keep the existing `instanceMatrix.needsUpdate = true` / `instanceColor.needsUpdate = true`
// at the end of syncInstances(). setMatrixAt/setColorAt never set those flags
// themselves — they just do a toArray() into the same buffer — so writing the
// floats directly is behaviourally identical.

push(e: InstancedAgent, playerX: number, playerZ: number): boolean {
    // ... existing setup per enemy, unchanged: cull / LOD pick / wrapped render
    //     position (rx, ry, rz) / `parts` selection / tint source ...

    const mtx = this.mesh.instanceMatrix.array as Float32Array;
    const col = this.mesh.instanceColor!.array as Float32Array;

    // Yaw trig once per enemy; the yaw matrix is gone entirely.
    const cy = Math.cos(e.yaw), sy = Math.sin(e.yaw);

    for (const p of parts) {
      // ... existing swing selection, unchanged: sw, pivot (ax, ay, az),
      //     half-extents (bx, by, bz), vertical offset `off`, colour r/g/b ...

      // FINAL = T(rx,ry,rz) · Ry(yaw) · T(ax,ay,az) · Rx(sw) · S(bx,by,bz)·T(0,-off,0)
      // expanded by hand. Rows: m11..m33 = rotation·scale, t* = local translation
      // before the outer yaw + world offset.
      let m11: number, m31: number,
          m12: number, m22: number, m32: number,
          m13: number, m23: number, m33: number,
          tx: number, ty: number, tz: number;

      if (sw === 0) {
        // Torso, head, hips, weapon on most frames: no swing -> sin/cos skipped.
        m11 = cy * bx;  m31 = -sy * bx;
        m12 = 0;        m22 = by;        m32 = 0;
        m13 = sy * bz;  m23 = 0;         m33 = cy * bz;
        tx = ax;        ty = ay - off;   tz = az;
      } else {
        const cs = Math.cos(sw), sn = Math.sin(sw);
        const syn = sy * sn, cyn = cy * sn;
        const sycs = sy * cs, cycs = cy * cs;
        m11 = cy * bx;    m31 = -sy * bx;
        m12 = syn * by;   m22 = cs * by;   m32 = cyn * by;
        m13 = sycs * bz;  m23 = -sn * bz;  m33 = cycs * bz;
        tx = ax;          ty = ay - cs * off;  tz = az - sn * off;
      }

      const bi = this.liveBody;
      const o = bi * 16;

      // three.js stores Matrix4 column-major: [c0 c1 c2 c3]
      mtx[o     ] = m11;  mtx[o +  1] = 0;    mtx[o +  2] = m31;  mtx[o +  3] = 0;
      mtx[o +  4] = m12;  mtx[o +  5] = m22;  mtx[o +  6] = m32;  mtx[o +  7] = 0;
      mtx[o +  8] = m13;  mtx[o +  9] = m23;  mtx[o + 10] = m33;  mtx[o + 11] = 0;
      mtx[o + 12] = rx + cy * tx + sy * tz;
      mtx[o + 13] = ry + ty;
      mtx[o + 14] = rz + cy * tz - sy * tx;
      mtx[o + 15] = 1;

      const c = bi * 3;
      col[c] = r; col[c + 1] = g; col[c + 2] = b;   // was setColorAt(bi, colour)

      this.liveBody = bi + 1;
    }

    return true;   // ... existing return value ...
}
```

**What changed**

- **The whole Matrix4 chain is gone.** `makeTranslation` / `makeRotationY` / `makeRotationX` / `compose` / 3× `multiplyMatrices` / `multiply` per part — each writing or reading 16 floats and doing 48 multiply-adds — collapse into the closed-form expansion above: **~11 scalar multiplies (8 with no swing) plus 16 stores per part**. At 96 enemies × 8 LOD0 parts that removes ~4,600 Matrix4 multiplies per frame.
- **`setMatrixAt` / `setColorAt` bypassed.** Both are just `toArray()` into `instanceMatrix.array` / `instanceColor.array`, so the direct writes are identical, minus the call, the bounds arithmetic and the Matrix4 round-trip.
- **Trig hoisted and halved.** Yaw `sin`/`cos` once per enemy (as before), and the swing `sin`/`cos` are skipped entirely for non-swinging parts — typically ~5 of 8 parts, i.e. ~1,000 fewer trig calls per frame.
- **Shared subexpressions reused:** `sy*sn`, `cy*sn`, `sy*cs`, `cy*cs` are each computed once and then scaled by `by`/`bz`.
- **No scratch objects touched.** `M_RENDER`, `M_YAW`, `M_OUTER`, `M_LOCAL`, `M_SWING`, `M_PIVOT`, `M_INNER`, `M_FINAL`, `vTrans`, `vScale`, `IDENTITY_QUAT` are all unused by this path now — delete them once no other method references them, and the per-frame cache pressure of streaming 8 Matrix4s per part goes with them.

**Optional extra 3,000 stores/frame:** lanes `o+3`, `o+7`, `o+11`, `o+15` are always `0,0,0,1` for every instance, forever. Initialise them once when the buffer is allocated (`for (let i = 0; i < max; i++) { a[i*16+15] = 1; }`, rest already zero) and delete those four assignments from the hot loop.

**Sanity check of the expansion** — with `sw = 0` the swing branch reduces to `cs = 1, sn = 0`, giving `m12 = m32 = m23 = 0`, `m22 = by`, `m13 = sy*bz`, `m33 = cy*bz`, `ty = ay - off`, `tz = az`, which is exactly the fast path. The two branches agree at the boundary.
