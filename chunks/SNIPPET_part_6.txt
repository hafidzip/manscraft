}

/**
 * One O(children) compaction instead of N × (indexOf + splice).
 * THREE.Object3D.remove() does children.indexOf() + splice(), so retiring 200
 * particles from a 3k-child scene is ~600k comparisons + 200 splices. This is one pass.
 */
function flushDetach(parent: THREE.Object3D): void {
  const n = DETACH.length;
  if (n === 0) return;
  if (n < 6) {                                  // tiny batches: plain remove is cheaper
    for (let i = 0; i < n; i++) parent.remove(DETACH[i]);
    DETACH.length = 0;
    return;
  }
  for (let i = 0; i < n; i++) (DETACH[i] as any).__fxDead = true;
  const kids = parent.children;
  let w = 0;
  for (let r = 0, m = kids.length; r < m; r++) {
    const o = kids[r] as any;
    if (o.__fxDead === true) {
      o.__fxDead = false;
      o.parent = null;
      o.dispatchEvent({ type: 'removed' });     // same event THREE.remove() fires
    } else {
      kids[w++] = kids[r];
    }
  }
  kids.length = w;
  DETACH.length = 0;
}

**What changed**
- All four `splice(i, 1)` calls → `swapPop` (swap with last, pop). Reverse iteration makes this provably safe: the element swapped into slot `i` came from an index `> i`, which was already processed this frame, and `i` then decrements past it — no double-tick, no skipped element. Worst case goes from O(N²) to O(N).
- `scene.remove()` per dead mesh → a single `flushDetach` pass. Kills the `indexOf` scan that dominated mass-death frames (grenades, cluster deaths).
- Optional `camPos` arg gates `world.solid()` bounce probes to a 28-unit bubble. Default-off, so existing callers get identical behaviour until you pass the camera.
- Hoisted `this.particles` / `this.casings` / … into locals so the loop doesn't reload the property each iteration.
- **Caveat:** swap-remove destroys insertion order. Harmless for additive/depth-tested FX; if tracers are ever alpha-blended and need painter-order sorting, keep `splice` for that one array only.
- **Next step (biggest remaining win):** keep pooled meshes permanently parented to a dedicated `THREE.Group` and toggle `mesh.visible` instead of add/remove. That removes scene-graph churn entirely and makes `flushDetach` unnecessary.

const KEY_Y_SHIFT = 9;
const KEY_Z_SHIFT = 17;

/**
 * Packs a world cell into a 26-bit non-negative SMI — zero allocation.
 * `x & 0x1FF` is exactly `x mod 512` for negative x too, so it matches the
 * torus world (WORLD_SIZE = 512) and cells across the seam now dedupe correctly.
 */
const cellKey = (x: number, y: number, z: number): number =>
  (x & 0x1FF) | ((y & 0xFF) << KEY_Y_SHIFT) | ((z & 0x1FF) << KEY_Z_SHIFT);

const KEY_EMPTY = -1;
const KEY_TOMB  = -2;

/** Open-addressed integer set: linear probing + tombstones. No per-op allocation. */
class IntSet {
  private keys: Int32Array;
  private mask: number;
  private limit: number;
  private live = 0;   // live entries
  private used = 0;   // live + tombstones

  constructor(capPow2 = 1024) {
    this.keys  = new Int32Array(capPow2).fill(KEY_EMPTY);
    this.mask  = capPow2 - 1;
    this.limit = (capPow2 * 3) >> 2;          // ≥25% of slots stay EMPTY → probes terminate
  }

  get size(): number { return this.live; }

  private slotOf(k: number, mask: number): number {
    return (Math.imul(k, 0x9E3779B1) >>> 15) & mask;   // Knuth multiplicative hash
  }

  /** Returns true if the key was newly inserted (has + add in a single probe). */
  add(k: number): boolean {
    const t = this.keys, m = this.mask;
    let i = this.slotOf(k, m);

      const v = t[i];

        if (tomb >= 0) { t[tomb] = k; }          // reuse tombstone: `used` unchanged

        this.live++;
        if (this.used >= this.limit) this.rehash();
        return true;
