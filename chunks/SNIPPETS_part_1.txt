## 0. Design summary

The whole brief collapses into **one primitive**: a 1-byte **origin tag** that answers
*"which planet did this block come from?"* Everything else hangs off it.

```
OriginTag : uint8
  bits 0-3  : planet type index + 1   (1..14, 0 = untagged/legacy)
  bits 4-7  : hue bucket 0..15        (derived from planet seed)
```

Why one byte matters:

| Consumer | What the tag buys |
|---|---|
| `core/textures.ts` | selects a **per-`PlanetType` atlas band** → real per-planet *shape* art (bark pattern, blade silhouette, petal count) |
| vertex `col` attribute | multiplies by **per-seed hue tint** → two `jungle` planets with different seeds look different |
| `world/blocks.ts` | `colorsFor(id, tag)` replaces the globally-mutated `DEFS[id].colors` |
| `fps/Inventory.ts` | `SlotItem.origin` → carried stacks keep their look and their name ("Jungle Wood Log") |
| `persist/worldDelta.ts` | second byte per voxel record → origin survives save/load (v1→v2 migration) |
| `crafting/recipes.ts` | irrelevant to matching (categories handle that) but **propagates** to outputs |

Two-part answer to the atlas problem, exactly as the brief's "(a) or (b)" note anticipated —
**do both**: `(a)` bands give *structure* per `PlanetType`; `(b)` vertex tint gives
*hue* per seed. Because hue is no longer baked into pixels, **the atlas becomes
theme-independent and is built exactly once and cached** — `createTextures()` stops being a
per-planet repaint, and two planets' logs can be on screen simultaneously.

## 1. `src/game/core/origin.ts` — NEW (the primitive)

Leaf module: imports **only** `space/palettes.ts`, so both block-id systems, the crafting
pipeline and the FPS inventory can import it with zero cycle risk.

```ts
// src/game/core/origin.ts
import { PLANET_PALETTES, rampAt255, type PlanetType } from '../space/palettes';

export type OriginTag = number; // uint8; 0 === untagged / legacy / non-themed
export type RGB = [number, number, number];
export type TintMap = Record<string, RGB>;

export const NO_ORIGIN: OriginTag = 0;

/** Canonical order — DO NOT REORDER: indices are baked into save files & atlas bands. */
export const PLANET_TYPE_ORDER = [
  'terran', 'ocean', 'desert', 'ice', 'volcanic', 'alien', 'barren',
  'jungle', 'savanna', 'tundra', 'lava', 'oceanic_ice', 'crimson', 'neon',
] as const satisfies readonly PlanetType[];

export const VARIANT_COUNT = PLANET_TYPE_ORDER.length; // 14 → fits the 4-bit nibble
const TYPE_INDEX = new Map<PlanetType, number>(PLANET_TYPE_ORDER.map((t, i) => [t, i]));

// ---------------------------------------------------------------- pack / unpack
export function packOrigin(type: PlanetType, hueBucket: number): OriginTag {
  const ti = TYPE_INDEX.get(type) ?? 0;
  const hb = ((hueBucket % 16) + 16) % 16;
  return (((ti + 1) & 0x0f) | (hb << 4)) & 0xff;
}
export const originTypeIndex = (tag: OriginTag): number => (tag & 0x0f) - 1;
export function originType(tag: OriginTag): PlanetType | null {
  const i = originTypeIndex(tag);
  return i >= 0 && i < VARIANT_COUNT ? PLANET_TYPE_ORDER[i] : null;
}
export const originHueBucket = (tag: OriginTag): number => (tag >> 4) & 0x0f;

/** Human label for inventory tooltips: "Jungle", "Oceanic Ice", … */
export function originLabel(tag: OriginTag): string | null {
  const t = originType(tag);
  return t ? PLANET_PALETTES[t].name : null;
}

/** Deterministic 4-bit hue bucket from a planet seed (splitmix64 finalizer). */
export function hueBucketFromSeed(seed: bigint): number {
  const M = 0xffffffffffffffffn;
  let h = BigInt.asUintN(64, seed);
  h = ((h ^ (h >> 30n)) * 0xbf58476d1ce4e5b9n) & M;
  h = ((h ^ (h >> 27n)) * 0x94d049bb133111ebn) & M;
  h ^= h >> 31n;
  return Number(h & 0x0fn);
}

/** Per-seed HSV nudge. Shared by theme.ts so theme.colors and voxel tints agree exactly. */
export function hueBucketOffsets(b: number): { hue: number; sat: number; val: number } {
  return {
    hue: (b / 16 - 0.47) * 0.11,          // ±~6% of the wheel
    sat: 0.86 + (b % 5) * 0.07,           // 0.86 .. 1.14
    val: 0.93 + ((b >> 2) & 3) * 0.045,   // 0.93 .. 1.065
  };
}

// ---------------------------------------------------- shared tint reference tables
// MOVED here from core/textures.ts so textures.ts, blocks.ts and origin.ts share one truth.
export const TINT_REF: TintMap = {
  grass: [96, 162, 54], dirt: [121, 85, 58], stone: [128, 128, 128], sand: [219, 207, 163],
  leaves: [56, 120, 40], log: [104, 76, 44], planks: [164, 129, 80], snow: [238, 246, 248],
  water: [58, 102, 222], cactus: [62, 138, 56],
};

/** Which height on the planet's `pal.stops` ramp represents each material. */
export const GROUP_H: Record<string, number> = {
  grass: 0.14, leaves: 0.22, dirt: 0.08, sand: 0.04,
  stone: 0.5, log: 0.3, planks: 0.36, water: -0.2, cactus: 0.18,
};

const TINT_STRENGTH = 0.85, TINT_LO = 0.22, TINT_HI = 2.4;
const clamp = (v: number, a: number, b: number) => (v < a ? a : v > b ? b : v);

export function groupTargetRgb(type: PlanetType, group: string): RGB {
  const pal = PLANET_PALETTES[type];
  if (group === 'snow') {
    return [pal.pole[0] * 255, pal.pole[1] * 255, pal.pole[2] * 255];
  }
  return rampAt255(pal.stops, GROUP_H[group] ?? 0.12);
}

// --- minimal RGB<->HSV so the per-seed nudge is perceptual, not per-channel mush -----
function rgb2hsv([r, g, b]: RGB): RGB {
  r /= 255; g /= 255; b /= 255;
  const mx = Math.max(r, g, b), mn = Math.min(r, g, b), d = mx - mn;
  let h = 0;
  if (d > 1e-6) {
    if (mx === r) h = ((g - b) / d + 6) % 6;
    else if (mx === g) h = (b - r) / d + 2;
    else h = (r - g) / d + 4;
    h /= 6;
  }
  return [h, mx <= 0 ? 0 : d / mx, mx];
}
function hsv2rgb([h, s, v]: RGB): RGB {
  const i = Math.floor(h * 6), f = h * 6 - i;
  const p = v * (1 - s), q = v * (1 - f * s), t = v * (1 - (1 - f) * s);
  const [r, g, b] = [[v, t, p], [q, v, p], [p, v, t], [p, q, v], [t, p, v], [v, p, q]][i % 6];
  return [r * 255, g * 255, b * 255];
}
function nudge(c: RGB, o: { hue: number; sat: number; val: number }): RGB {
  const [h, s, v] = rgb2hsv(c);
  return hsv2rgb([(h + o.hue + 1) % 1, clamp(s * o.sat, 0, 1), clamp(v * o.val, 0, 1)]);
}
