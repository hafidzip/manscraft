## 0. Load-bearing decisions (locked before any code)

| # | Decision |
|---|---|
| D1 | Permanence is **not** `hardness = Infinity`. Ores keep finite hardness (= harvest duration) and are special-cased at *cycle completion* via one authoritative predicate `isOreBlock(id)` in `src/game/world/blocks.ts`. |
| D2 | One shared harvest rule, `harvestOutcome(worldId)`, returns `{ itemId, destroy }`. Ore ⇒ `destroy: false`. Player mining, live `LaserMiner`, and headless `factorySim` all call it, so they cannot drift. |
| D3 | Laser miners acquire exactly `(wrap(x+dx), y, wrap(z+dz))`. No cone, no range, no lateral/vertical, no pass-through. Acquisition is O(1), so re-acquire every cycle is free. |
| D4 | Raw coal ore gets a **new** FPS/item id `65` (`B.RAW_COAL_ORE`). Processed coal stays `58`. `TO_FPS[world.COAL_ORE] = 65`. |
| D5 | Underground = `oreMaxY = min(h - ORE_SURFACE_CLEARANCE, sea - 2, H - 8)` with `ORE_SURFACE_CLEARANCE = 6`, applied to **every** ore including coal. Depth bands are rebalanced to fit under `sea - 2 = 28`. |
| D6 | Furnace results are **ejected immediately** as drops/ledger units on top of the furnace. `st.output` survives only as a legacy field that is drained once on load and never refilled, so a full output slot can never stall automation. |
| D7 | Inserter → furnace uses **relevant-slot** rejection and never merges. Inserter → crafting table accepts only ingredients of the persisted blueprint, below the per-craft required count. |
| D8 | Crafting blueprint + ingredient buffer live in `CraftingTableState`, owned by `PlanetFactorySim`, shared by reference with `GameEngine`, persisted per planet. React holds only a mirror. |
| D9 | Terrain determinism changed ⇒ `SAVE_VERSION 1 → 2`. v1 planet records are archived (not silently replayed) and the planet regenerates. |
| D10 | Every unit lives in exactly one place. Transfers are `take → (insert | emit)` with no path that returns without doing exactly one of the two. |

## 1. NEW MODULE — `src/game/factory/machineProcessing.ts` (complete source)

Pure TypeScript. **No `three`, no DOM, no engine imports.** This is the single shared automation layer.

```ts
/**
 * src/game/factory/machineProcessing.ts
 *
 * Shared, pure machine-automation rules. Live scene (GameEngine + InserterManager)
 * and headless PlanetFactorySim MUST both route through this module so acceptance
 * behavior cannot diverge between rendered and simulated planets.
 *
 * Invariants enforced here:
 *  - one unit is accepted into machine state, or it is rejected (caller emits it) - never both
 *  - furnace insertion never merges into an occupied *relevant* slot
 *  - crafting insertion only accepts blueprint ingredients below their per-craft count
 *  - a completed craft consumes exactly one ingredient multiset and returns exactly one output
 */

import type { SlotItem } from '../fps/Inventory';
import { wrapBlock } from '../core/constants';
import { isOreBlock } from '../world/blocks';
import { type FurnaceState, isFuel, smeltResult } from '../crafting/smelting';
import { RECIPES, recipeIngredients, type Recipe } from '../crafting/recipes';

/* ------------------------------------------------------------------ keys */

/** Same string shape as the pre-existing furnaceKey(); machines share one keyspace shape. */
export const machineKey = (x: number, y: number, z: number): string =>
  `${wrapBlock(x | 0)},${y | 0},${wrapBlock(z | 0)}`;

export function parseMachineKey(k: string): [number, number, number] | null {
  const p = k.split(',');
  if (p.length !== 3) return null;
  const x = Number(p[0]);
  const y = Number(p[1]);
  const z = Number(p[2]);
  if (!Number.isFinite(x) || !Number.isFinite(y) || !Number.isFinite(z)) return null;
  return [wrapBlock(x | 0), y | 0, wrapBlock(z | 0)];
}

/* ----------------------------------------------------------------- sinks */

/** One unit of an FPS item id. Used for spills and produced results. */
export interface StackOut {
  id: number;
  count: number;
}

/**
 * Abstract "put items into the world above this machine".
 * Live  -> ItemDropManager.spawn(itemId, new THREE.Vector3(x+0.5, y+1.15, z+0.5))
 * Headless -> ItemLedger.addAtCell(packCell(x, y+1, z), itemId, count)
 */
export interface MachineItemSink {
  emitAbove(x: number, y: number, z: number, itemId: number, count?: number): void;
}

/* ---------------------------------------------------- exact-front target */

export interface VoxelRead {
  getBlock(x: number, y: number, z: number): number;
}

export interface FrontCell {
  x: number;
  y: number;
  z: number;
}

/** The one and only cell a machine may act on: adjacent, same Y, facing direction. */
export function frontCellOf(x: number, y: number, z: number, dx: number, dz: number): FrontCell {
  return { x: wrapBlock(x + dx), y, z: wrapBlock(z + dz) };
}

/**
 * Exact-front acquisition shared by live LaserMiner and headless factorySim.
 * Returns the front cell only when `mineable(id)` accepts the block there.
 */
export function acquireFrontTarget(
  view: VoxelRead,
  x: number,
  y: number,
  z: number,
  dx: number,
  dz: number,
  mineable: (id: number) => boolean,
): { cell: FrontCell; id: number } | null {
  const cell = frontCellOf(x, y, z, dx, dz);
  const id = view.getBlock(cell.x, cell.y, cell.z);
  if (id < 0 || !mineable(id)) return null;
  return { cell, id };
}

/* --------------------------------------------------------- harvest rule */

export interface HarvestOutcome {
  /** FPS item id to emit exactly once. */
  itemId: number;
  /** false for permanent infinite ore nodes; true for ordinary finite blocks. */
  destroy: boolean;
}

/**
 * The single completion rule for ALL mining paths (player laser, live placed miner,
 * headless placed miner). `toFps` is engine/constants TO_FPS, injected to keep this
 * module free of engine imports.
 */
export function harvestOutcome(
  worldId: number,
  toFps: Record<number, number>,
): HarvestOutcome | null {
  const itemId = toFps[worldId];
  if (itemId === undefined) return null;
  return { itemId, destroy: !isOreBlock(worldId) };
}

/* --------------------------------------------------------------- furnace */

export type FurnaceInsertResult = 'input' | 'fuel' | 'rejected';

/**
 * Relevant-slot insertion. Never merges, even for identical items.
 *  smeltable + empty input -> 'input'
 *  smeltable + busy  input -> 'rejected'   (fuel slot is irrelevant here)
 *  fuel      + empty fuel  -> 'fuel'
 *  fuel      + busy  fuel  -> 'rejected'
 *  neither                 -> 'rejected'
 * Classification order matches the existing quick-move code: smeltable first, then fuel.
 */
export function tryInsertFurnace(st: FurnaceState, itemId: number): FurnaceInsertResult {
  if (!Number.isFinite(itemId) || itemId <= 0) return 'rejected';
  if (smeltResult(itemId)) {
    if (st.input) return 'rejected';
    st.input = { kind: 'block', blockId: itemId, count: 1 };
    return 'input';
  }
  if (isFuel(itemId)) {
    if (st.fuel) return 'rejected';
    st.fuel = { kind: 'block', blockId: itemId, count: 1 };
    return 'fuel';
  }
  return 'rejected';
}

/** Everything a broken furnace owes the world. Called exactly once, then delete the state. */
export function spillFurnace(st: FurnaceState): StackOut[] {
  const out: StackOut[] = [];
  const push = (s: SlotItem | null): void => {
    if (s && s.kind === 'block' && s.count > 0) out.push({ id: s.blockId, count: s.count });
  };
  push(st.input);
  push(st.fuel);
  push(st.output); // legacy pre-migration output only
  st.input = null;
  st.fuel = null;
