### 4.3 World storage & change events — `src/game/world/world.ts`

The single source of voxel truth. **Feature A hooks in here** (`ensureData` is where generated
chunk data appears → replay deltas there; `setBlock`/`destroySphere` is where edits are made →
record deltas via `onChanged`).

```ts
interface Chunk {
  cx: number; cz: number;
  data: Uint8Array;            // S*H*S = 16*80*16 = 20 480 bytes
  meshes: THREE.Mesh[]; hasMesh: boolean; grass: THREE.Mesh | null;
  kx: number; kz: number; dirty: boolean;
  colH: Uint8Array; colC: Uint32Array; colW: Uint8Array;   // cached column height/color/water
}

export class World {
  readonly group = new THREE.Group();
  readonly gen: TerrainGenerator;
  private chunks = new Map<number, Chunk>();
  private fullyPrepared = false;      // after prepareAllData(): data is NEVER evicted
  // …queues: loadQueue/dirtyQueue (meshing), evictIter

  onChanged: ((x, y, z, oldId, newId) => void) | null = null;
  onChunkData: ((cx, cz, data: Uint8Array, fromGenerator: boolean) => void) | null = null;
  onChunkGone: ((cx, cz) => void) | null = null;
  spawn = new THREE.Vector3(8.5, 45, 8.5);

  constructor(public readonly seed: number, private mats: ChunkMaterials) {
    this.gen = new TerrainGenerator(seed);
  }

  private key(cx: number, cz: number): number { return (wrapChunk(cx) << 5) | wrapChunk(cz); }

  forEachLoadedChunk(cb: (cx: number, cz: number, data: Uint8Array) => void): void {
    for (const c of this.chunks.values()) cb(c.cx, c.cz, c.data);
  }

  private ensureData(rawCx: number, rawCz: number): Chunk {
    const cx = wrapChunk(rawCx), cz = wrapChunk(rawCz);
    const k = (cx << 5) | cz;
    if (k === this.memoKey && this.memoChunk) return this.memoChunk;
    let c = this.chunks.get(k);
    if (!c) {
      const data = new Uint8Array(S * H * S);
      this.gen.populateChunk(data, cx, cz);          // ← deterministic base terrain
      c = { cx, cz, data, meshes: [], hasMesh: false, grass: null, kx: 0, kz: 0,
            dirty: false, colH: new Uint8Array(S*S), colC: new Uint32Array(S*S),
            colW: new Uint8Array(S*S) };
      this.chunks.set(k, c);
      this.buildColumnCache(c);
      // ▼▼▼ CRITICAL INJECTION POINT (Feature A): apply persisted per-chunk deltas to
      // `c.data` HERE, BEFORE this notification fires (registry rescans on this signal).
      this.onChunkData?.(cx, cz, c.data, true);
      if (!this.bulkPreparing) this.markNeighborBorders(cx, cz);
    }
    this.memoKey = k; this.memoChunk = c;
    return c;
  }

  peekBlock(wx, wy, wz): number {            // -1 when chunk data absent; NO auto-generate
    if (wy < 0) return B.BEDROCK;
    if (wy >= H) return B.AIR;
    const px = Math.floor(wrapBlock(wx)), pz = Math.floor(wrapBlock(wz));
    const c = this.peekChunk(Math.floor(px / S), Math.floor(pz / S));
    if (!c) return -1;
    return c.data[chunkIndex(px % S, wy, pz % S)];
  }

  getBlockRaw(wx, wy, wz): number { /* same but this.ensureData(...) → generates on demand */ … }
  get(x,y,z) { return this.getBlockRaw(x,y,z); }
  set(x,y,z,id) { this.setBlock(x,y,z,id); }

  setBlock(wx, wy, wz, id): void {
    if (wy < 0 || wy >= H) return;
    const px = Math.floor(wrapBlock(wx)), pz = Math.floor(wrapBlock(wz));
    const cx = Math.floor(px / S), cz = Math.floor(pz / S);
    const lx = px % S, lz = pz % S;
    const c = this.ensureData(cx, cz);
    const oldId = c.data[chunkIndex(lx, wy, lz)];
    if (oldId === id) return;
    c.data[chunkIndex(lx, wy, lz)] = id;
    this.updateColumn(c, lx, lz, wy, id);
    if (this.batchDepth === 0) {
      c.dirty = true;
      this.onChanged?.(px, wy, pz, oldId, id);       // ← every live edit funnels through here
      this.buildMesh(c);
    }
    if (lx === 0) this.markDirty(cx - 1, cz);        // …plus 3 other borders
    if (this.batchDepth > 0) this.markDirty(cx, cz);
  }

  destroySphere(center, radius, onBlockDestroyed?): number {
    // …carves blocks with setBlock under beginBatch()/endBatch(), then re-meshes touched
    // chunks, then fires this.onChanged per carved cell: notify(px, y, pz, oldId, B.AIR)
  }

  // update(): per-frame budgeted meshing of dirtyQueue/loadQueue + eviction scan.
  // Chunks farther than EVICT_DISTANCE lose MESHES; data is dropped ONLY before
```

### 4.4 Terrain generation — `src/game/world/generator.ts`

Deterministic from the planet seed ⇒ **persist only deltas, never base voxels.**

```ts
export interface PlanetTheme {
  seed: bigint | string | number;
  name?: string; biome?: Biome | null;
  hillAmp?: number; mountainAmp?: number; seaLevel?: number; oceanCoverage?: number;
  [k: string]: unknown;
}
export function planetSeedToWorldSeed(seed: bigint | string | number): number {
  const b = typeof seed === 'bigint' ? seed : BigInt(seed);
  const s = b < 0n ? -b : b;
  const lo = Number(s & 0xffffffffn), hi = Number((s >> 32n) & 0xffffffffn);
  return ((lo ^ Math.imul(hi, 0x9e3779b1)) >>> 0) % 0x7fffffff || 1;
}
```

### 4.5 ChangeBus — `src/game/world/changeBus.ts` (complete)

The broadcast hub. **Feature A's recorder attaches here**; note sink ORDER matters —
the delta-replay sink must run before `MachineRegistry`'s rescan sink, so apply deltas
inside `World.ensureData` (preferred) rather than as a bus sink.

```ts
export type BlockChangeFn = (x, y, z, oldId, newId) => void;
export type ChunkDataFn = (cx, cz, data: Uint8Array, fromGenerator: boolean) => void;
export type ChunkGoneFn = (cx, cz) => void;
export interface ChangeSink { onBlock?: BlockChangeFn; onChunkData?: ChunkDataFn; onChunkGone?: ChunkGoneFn; }

export class ChangeBus {
  version = 1;
  readonly stats = { blocks: 0, chunkData: 0, chunkGone: 0 };
  private sinks: ChangeSink[] = [];
  add(sink: ChangeSink): () => void { /* returns detach */ }
  attach(world: ChangeSource): void {
    const w = world as ChangeSource & { [ATTACHED]?: boolean };
    if (w[ATTACHED]) return;
    w[ATTACHED] = true;
    const prevBlock = world.onChanged;
    world.onChanged = (x, y, z, oldId, newId) => {
      prevBlock?.(x, y, z, oldId, newId);
      this.version++; this.stats.blocks++;
      for (let i = 0; i < this.sinks.length; i++) this.sinks[i].onBlock?.(x, y, z, oldId, newId);
    };
    const prevData = world.onChunkData ?? null;
    world.onChunkData = (cx, cz, data, fromGenerator) => {
      prevData?.(cx, cz, data, fromGenerator);
      this.stats.chunkData++;
      for (let i = 0; i < this.sinks.length; i++) this.sinks[i].onChunkData?.(cx, cz, data, fromGenerator);
    };
    // …same wrapping for onChunkGone
  }
}
```
