  // ---------------------------------------------------------------------------
  // Torus handling
  // ---------------------------------------------------------------------------
  /** Keep the ship in the player's min-image continuum (render space). Cheap. */
  rebase(refX: number, refZ: number): void {
    const nx = refX + minImageF(this.pos.x - refX);
    const nz = refZ + minImageF(this.pos.z - refZ);
    if (nx === this.pos.x && nz === this.pos.z) return;
    this.pos.x = nx;
    this.pos.z = nz;
    this.sync();
  }

  distanceTo(p: THREE.Vector3): number {
    return Math.hypot(minImageF(this.pos.x - p.x), this.pos.y - p.y, minImageF(this.pos.z - p.z));
  }

  // ---------------------------------------------------------------------------
  // Persistence
  // ---------------------------------------------------------------------------
  getTransform(): ShipTransform {
    return { x: wrapBlock(this.pos.x), y: this.pos.y, z: wrapBlock(this.pos.z), yaw: this.yaw };
  }

  setTransform(t: ShipTransform, refX: number, refZ: number): void {
    this.pos.set(refX + minImageF(t.x - refX), t.y, refZ + minImageF(t.z - refZ));
    this.yaw = t.yaw;
    this.vel.set(0, 0, 0);
    this.bank = 0;
    this.pitchVis = 0;
    this.flyPitch = 0;
    this.baseY = t.y;
    this.settleT = 0;
    this.settleHere(true);              // snap-clamp: never restore inside terrain
  }
```
