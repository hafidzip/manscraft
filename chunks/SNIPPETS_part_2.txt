      if (!e.alive) continue;
      e.tier = this.tierFor(e);
      if (e.tier === TIER_HOT) s.hot++;
      else if (e.tier === TIER_WARM) s.warm++;
      else if (e.tier === TIER_COLD) s.cold++;
      else s.dormant++;
      if (scaleFlags.renderLod) e.setAsleep(e.distToPlayer > this.renderCutoff);
      if (!e.asleep) s.rendered++;
    }
  }

  /** Hysteresis: promote at the cutoff, demote only past cutoff * TIER_HYST.
   *  A tier may therefore overshoot its budget by the hysteresis margin —
   *  intentional, and the reason §8.2 accepts hot <= HOT_BUDGET * 1.25. */
  private tierFor(e: Enemy): Tier {
    const d = e.distToPlayer;
    if (e.cfg.peaceful) {
      return d < HOT_R ? TIER_HOT : d < COLD_R ? TIER_WARM : TIER_COLD;
    }
    const h = TIER_HYST;
    if (d < this.hotCutoff  || (e.tier === TIER_HOT  && d < this.hotCutoff  * h)) return TIER_HOT;
    if (d < this.warmCutoff || (e.tier === TIER_WARM && d < this.warmCutoff * h)) return TIER_WARM;
    if (d < this.coldCutoff || (e.tier === TIER_COLD && d < this.coldCutoff * h)) return TIER_COLD;
    return TIER_DORMANT;
  }
  ```

  ```ts
  update(dt: number) {
    if (!this.enabled) return;
    pathBudget.tokens = pathBudget.maxTokens;

    const ppx = this.player.pos.x, ppz = this.player.pos.z;
    this.assignTiers(ppx, ppz);
    if (scaleFlags.grid) this.grid.rebuild(this.enemies);   // Phase 3

    const active = this.activeScratch;
    active.length = 0;

    for (let i = this.enemies.length - 1; i >= 0; i--) {
      const e = this.enemies[i];
      if (!e.alive) { /* existing corpse / cleanup path */ continue; }

      // Bank real time for EVERY enemy on EVERY frame. An agent that ticks at
      // 4 Hz must still receive a full quarter-second of movement.
      e.tickAccum += dt;
      const period = scaleFlags.tiers ? TIER_PERIOD_S[e.tier] : 0;
      if (e.tickAccum < period) continue;

      const sdt = e.tickAccum;
      e.tickAccum = 0;
      this.stats.ticks++;

      // Beyond SIM_RADIUS there is no streamed terrain to collide with, so run
      // a cheap kinematic advance instead of skipping the agent entirely.
      if (e.tier === TIER_DORMANT) { this.coarseAdvance(e, sdt, ppx, ppz); continue; }

      if (!this.tickEnemy(e, i, sdt)) continue;      // despawns on !keep
      if (e.alive && e.tier <= TIER_COLD) active.push(e);
    }

    this.separate(active, dt);                        // Phase 3
    // ... existing camp / respawn / merchant tails, unchanged
  }

  /** Spend banked time in bounded substeps: a 266 ms catch-up must not let an
   *  agent pass through a one-block floor. v1 instead clamped the dt, which is
   *  a silent slow-motion bug dressed up as a tunnelling fix. */
  private tickEnemy(e: Enemy, i: number, budget: number): boolean {
    const n = Math.min(MAX_SUBSTEPS, Math.max(1, Math.ceil(budget / MAX_TICK_DT)));
    const sdt = Math.min(budget / n, MAX_TICK_DT);
    this.stats.dtDropped += Math.max(0, budget - n * sdt);   // must stay ~0
    for (let s = 0; s < n; s++) {
      if (!e.update(sdt, this.player)) { this.despawn(i); return false; }
    }
    return true;
  }

  /** Off-screen advance: no collision, no pathing, no animation, ~0.2 us.
   *  Dormant agents can pass through terrain — they are outside the streamed
   *  region, so nothing observes them. Revisit if view distance ever exceeds
   *  SIM_RADIUS. */
  private coarseAdvance(e: Enemy, dt: number, ppx: number, ppz: number) {
    if (e.cfg.peaceful) return;
    const dx = minImageF(ppx - e.pos.x, WORLD_SIZE);
    const dz = minImageF(ppz - e.pos.z, WORLD_SIZE);
    const d = Math.sqrt(dx * dx + dz * dz) || 1;
    const step = e.cfg.speed * DORMANT_SPEED_SCALE * dt;
    e.pos.x = wrapCoord(e.pos.x + (dx / d) * step, WORLD_SIZE);
    e.pos.z = wrapCoord(e.pos.z + (dz / d) * step, WORLD_SIZE);
  }
