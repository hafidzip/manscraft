```ts
/** Relocate > reuse > allocate, in that order of preference. */
private placeWild(preset: string, x: number, y: number, z: number) {
  // 1. At the cap: move a genuinely distant wild agent. Never a camp member
  //    (it owns a home), never anything inside RELOCATE_R (it would be seen).
  if (this.hostiles >= MAX_POP) {
    const r = this.findRelocatable();
    if (!r) { this.refusals++; return; }   // everyone is close: refuse, don't teleport
    r.recycleTo(x, y, z);
    r.wild = true;
    r.alert(this._sv.set(this.player.pos.x, this.player.pos.y, this.player.pos.z));
    this.gridStale = true;
    return;
  }

  // 2. Below the cap: reuse a banked instance before constructing one.
  let e = this.pool.take(preset);
  if (e) {
    e.recycleTo(x, y, z);
  } else {
    // Fresh Vector3 here on purpose: if the ctor stores the reference rather
    // than copying, a shared scratch would alias every enemy to one position.
    e = new Enemy(preset, new THREE.Vector3(x, y, z), this.deps, {
      behavior: Math.random() < 0.5 ? 'patrol' : 'idle',
    });
    e.poolKey = preset;
  }
  e.wild = true;
  e.home.x = x; e.home.z = z;
  this.enemies.push(e);
  if (this.sceneRef) e.attachToScene(this.sceneRef);   // Phase 1 contract
  e.alert(this._sv.set(this.player.pos.x, this.player.pos.y, this.player.pos.z));
  this.hostiles++;
  this.gridStale = true;
}

/** O(1) amortised via a rotating cursor — not an O(N) farthest-scan per spawn. */
private findRelocatable(): Enemy | null {
  const n = this.enemies.length;
  if (n === 0) return null;
  for (let k = 0; k < n; k++) {
    const i = (this.recycleCursor + k) % n;
    const e = this.enemies[i];
    if (e.alive && e.wild && !e.cfg.peaceful && e.d2 > RELOCATE_R_SQ) {
      this.recycleCursor = (i + 1) % n;
      return e;
    }
  }
  return null;
}
```

### 5.6 Despawn folded into Phase 1's eviction pass

```ts
// Call once per frame, after the Phase 2 tier pass has written every e.d2.
private evictAndCull() {
  for (let i = this.enemies.length - 1; i >= 0; i--) {
    const e = this.enemies[i];
    const d2 = e.d2;                                  // already computed
    if (!e.alive && e.deathT > CORPSE_TTL)      { this.remove(i); continue; }
    if (e.alive && e.wild && d2 > DESPAWN_R_SQ) { this.remove(i); continue; }
    if (d2 > this.RENDER_RADIUS_SQ) e.detachFromScene();
    else if (!e.inScene && this.sceneRef) e.attachToScene(this.sceneRef);
  }
}

/** Swap-remove: O(1) vs splice's O(N). Correct only in a reverse loop —
 *  the element swapped in comes from an index already visited. */
private remove(i: number) {
  const e = this.enemies[i];
  const last = this.enemies.length - 1;
  this.enemies[i] = this.enemies[last];
  this.enemies.pop();
  if (!e.cfg.peaceful) this.hostiles--;
  this.pool.give(e, POOL_MAX);
  this.gridStale = true;
}
```

## Phase 6 — GPU Instancing (one draw call for the whole crowd)

**Effort:** 3–4 d (the doc's 5 d assumed skinned instancing; box-part instancing is cheaper)
**Capacity after:** 2,500 → 5,000+ — rendering stops being the wall, `Enemy.update()` becomes it
**New file:** `src/game/fps/EnemyInstancer.ts`
**Touches:** `src/game/fps/EnemyManager.ts`, `src/game/fps/Enemy.ts`, `src/game/engine.ts`

### Assumptions to check before you paste

1. `Enemy.pos` is the **feet** position (phase 5 spawns at `new THREE.Vector3(spawnX, h, spawnZ)` with `h = highestY`, so this holds). If it's the centre, subtract half-height once in `syncInstances`.
2. `Enemy` exposes a yaw scalar. If the yaw only lives in `group.rotation.y`, hoist it to `this.yaw` first — the group is going away.
3. Enemy materials are `MeshLambertMaterial`/`MeshStandardMaterial`. Set `material.color` to white; all tint moves to `instanceColor`, which three multiplies in.
4. three.js ≥ r140. `addUpdateRange` (r159+) is handled by a shim below.
5. `BatchedMesh` is *not* what you want here — it exists for **differing** geometries. Identical boxes → `InstancedMesh` wins.

### The six real bugs to avoid

1. **Negative scale flips winding.** Mirroring the right arm with `sx = -0.22` renders it inside-out under backface culling. Mirror with the pivot's sign, never with the scale.
2. **`frustumCulled` left `true`.** The instanced mesh sits at the origin with an identity/stale bounding sphere; look away from world-origin and the *entire crowd* vanishes. Set `frustumCulled = false` — culling one object saves nothing anyway.
3. **Stale `count` / missing `needsUpdate`.** `count` above capacity draws garbage; a forgotten `needsUpdate` freezes 2,500 enemies mid-stride. Clamp on push, mark on end.
4. **Torus seam.** `group.position.copy(e.pos)` renders a wrapped neighbour on the far side of the world — a latent bug in the current code that instancing lets you fix for free. Write instances at *minimum-image* positions: `player + wrapDelta(e.pos − player)`.
5. **Transparency can't be per-instance sorted.** If death is an opacity fade, convert it to sink + scale-to-zero, or hand dying agents to the small "detailed" pool. Keep the instanced material opaque.
6. **`InstancedMesh.raycast` is O(count) per ray.** No-op it and route all hit-testing through the phase 3 grid.
