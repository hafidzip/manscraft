  };

  private readonly slots: Slot[] = [];
  private readonly byKind: (Slot | null)[] = new Array(16).fill(null);
  private readonly view: MachineView =
    { ix: 0, iz: 0, d2: 0, px: 0, py: 0, pz: 0 };

  private unionRadius = 0;
  private unionKinds = 0;
  private unionYLoRel = 0;
  private unionYHiRel = 0;

  private px = 0; private py = 0; private pz = 0;
  private scanClock = 0;
  private dormantClock = 0;
  private detach: (() => void) | null = null;

  /**
   * `agents` order IS the old update order. engine.ts registers
   * [inserters, laserMiners, turrets], which reproduces the historic
   * itemDrops -> inserters -> laserMiners -> turrets frame sequence.
   */
  constructor(
    private readonly registry: MachineRegistry,
    agents: MachineAgent[],
  ) {
    for (const agent of agents) {
      const k = agent.maxLive;
      const slot: Slot = {
        agent,
        kind: agent.kind,
        keys: new Int32Array(k),
        d2: new Float64Array(k),
        ix: new Float64Array(k),
        iz: new Float64Array(k),
        recs: new Array(k).fill(null),
        n: 0,
        prevKeys: new Int32Array(k),
        prevN: 0,
        ownedKeys: [],
        ownedIdx: new Map(),
        pruneCursor: 0,
        thinkCursor: 0,
        absYLo: 0,
        absYHi: 0,
        r2: agent.scanRadius * agent.scanRadius,
        pruneR2: agent.pruneRadius * agent.pruneRadius,
        legacyReadsPerScan:
          discCells(agent.scanRadius) * (agent.yHi - agent.yLo + 1),
      };
      this.slots.push(slot);
      this.byKind[agent.kind] = slot;
      this.unionKinds |= agent.kind;
      if (agent.scanRadius > this.unionRadius) this.unionRadius = agent.scanRadius;
      if (agent.yLo < this.unionYLoRel) this.unionYLoRel = agent.yLo;
      if (agent.yHi > this.unionYHiRel) this.unionYHiRel = agent.yHi;
    }

    // A machine block that disappears must destroy its entity in the SAME frame,
    // which is exactly what the old `id >= 0 && !isInserter(id)` branch did.
    this.detach = registry.addListener({
      onRemove: (rec) => this.forget(rec.key, rec.kind),
      onIdChanged: (rec, oldId) => {
        this.byKind[rec.kind]?.agent.onIdChanged?.(rec, oldId);
      },
    });
  }

  dispose(): void { this.detach?.(); this.detach = null; }

  /** Pre-bound: forEachNear() receives the same function object every frame. */
  private visit = (rec: MachineRecord, ix: number, iz: number,
                   d2: number): void => {
    if (!rec.live) return;                      // dormant == old `id < 0` == hidden
    const slot = this.byKind[rec.kind];
    if (slot === null) return;
    if (d2 > slot.r2) return;                   // per-agent radius
    if (rec.y < slot.absYLo || rec.y > slot.absYHi) return;   // per-agent Y band
    this.insertTopK(slot, rec, ix, iz, d2);
  };

  /** Bounded insertion sort into a K-length ascending array. K <= 24. */
  private insertTopK(slot: Slot, rec: MachineRecord,
                     ix: number, iz: number, d2: number): void {
    const cap = slot.keys.length;
    let i: number;
    if (slot.n < cap) {
      i = slot.n++;
    } else {
      if (d2 >= slot.d2[cap - 1]) return;       // farther than the current worst
      i = cap - 1;
    }
    while (i > 0 && slot.d2[i - 1] > d2) {
      slot.keys[i] = slot.keys[i - 1];
      slot.d2[i]   = slot.d2[i - 1];
      slot.ix[i]   = slot.ix[i - 1];
      slot.iz[i]   = slot.iz[i - 1];
      slot.recs[i] = slot.recs[i - 1];
      i--;
    }
    slot.keys[i] = rec.key;
    slot.d2[i]   = d2;
    slot.ix[i]   = ix;
    slot.iz[i]   = iz;
    slot.recs[i] = rec;
  }

  update(dt: number, playerPos: THREE.Vector3): void {
    this.stats.frames++;
    this.registry.tickClock(dt);

    const px = playerPos.x, py = playerPos.y, pz = playerPos.z;
    this.px = px; this.py = py; this.pz = pz;
    this.view.px = px; this.view.py = py; this.view.pz = pz;

    const cy = Math.floor(py);
    const slots = this.slots;

    // ---- ONE shared proximity pass: 25 bucket probes, zero voxel reads ----------
    for (let s = 0; s < slots.length; s++) {
      const sl = slots[s];
      sl.n = 0;
      sl.absYLo = Math.max(0, cy + sl.agent.yLo);
      sl.absYHi = Math.min(WORLD_HEIGHT - 1, cy + sl.agent.yHi);
    }
    const yLo = Math.max(0, cy + this.unionYLoRel);
    const yHi = Math.min(WORLD_HEIGHT - 1, cy + this.unionYHiRel);
    this.registry.forEachNear(
      px, pz, this.unionRadius, yLo, yHi, this.unionKinds, this.visit,
    );

    // ---- instrumentation: what the three legacy scans would have cost -----------
    this.scanClock += dt;
    while (this.scanClock >= 0.5) {
      this.scanClock -= 0.5;
      let reads = 0;
      for (let s = 0; s < slots.length; s++) reads += slots[s].legacyReadsPerScan;
      this.stats.voxelReadsAvoided += reads;
      this.registry.stats.voxelReadsAvoided += reads;
    }

    let createBudget = CREATE_BUDGET_PER_FRAME;
    let activeTotal = 0;

    for (let s = 0; s < slots.length; s++) {
      const sl = slots[s];
      const agent = sl.agent;

      agent.prepare?.(dt, px, py, pz, sl.n > 0);

      // ---- creation, nearest-first, budgeted ----------------------------------
      for (let i = 0; i < sl.n; i++) {
        const key = sl.keys[i];
        if (agent.has(key)) continue;
        if (createBudget <= 0) {
          sl.recs[i] = null;                    // not ready: skip tick this frame
          this.stats.createDeferred++;
          continue;
        }
        const rec = sl.recs[i]!;
        agent.create(rec);
        this.remember(sl, key);
        createBudget--;
        this.stats.created++;
      }

      // ---- activation diff: only changed keys touch setActive() ---------------
      for (let i = 0; i < sl.n; i++) {
        if (sl.recs[i] === null) continue;
        const key = sl.keys[i];
        let wasActive = false;
        for (let j = 0; j < sl.prevN; j++) {
          if (sl.prevKeys[j] === key) { wasActive = true; break; }
        }
        if (!wasActive) agent.setActive(key, true);
      }
      for (let j = 0; j < sl.prevN; j++) {
        const key = sl.prevKeys[j];
        let stillActive = false;
        for (let i = 0; i < sl.n; i++) {
          if (sl.keys[i] === key && sl.recs[i] !== null) { stillActive = true; break; }
        }
        if (!stillActive && agent.has(key)) agent.setActive(key, false);
      }

      // ---- tick every active machine -----------------------------------------
      const view = this.view;
      for (let i = 0; i < sl.n; i++) {
        const rec = sl.recs[i];
        if (rec === null) continue;
        view.ix = sl.ix[i]; view.iz = sl.iz[i]; view.d2 = sl.d2[i];
        agent.tick(rec, view, dt);
        this.stats.ticks++;
      }

      // ---- think: round-robin, hard budget per frame --------------------------
      const budget = agent.thinkPerFrame;
      if (budget > 0 && sl.n > 0 && agent.think) {
        let done = 0;
        for (let a = 0; a < sl.n && done < budget; a++) {
