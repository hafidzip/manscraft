### 13.1 `src/game/persist/worldDelta.ts` — crafting tuples with validation

```diff
+import {
+  sanitizeCraftingState, machineKey, parseMachineKey,
+  type CraftingTableState,
+} from '../factory/machineProcessing';
+
+/** [x, y, z, recipeId, itemId1, count1, itemId2, count2, ...] */
+export type CraftingTuple = [number, number, number, string | null, ...number[]];
+
+export function serializeCraftingTables(m: Map<string, CraftingTableState>): CraftingTuple[] {
+  const out: CraftingTuple[] = [];
+  for (const [k, st] of m) {
+    const at = parseMachineKey(k);
+    if (!at) continue;
+    const pairs: number[] = [];
+    for (const key of Object.keys(st.buffered)) {
+      const id = Number(key);
+      const n = st.buffered[id] ?? 0;
+      if (id > 0 && n > 0) pairs.push(id, n);
+    }
+    if (!st.recipeId && pairs.length === 0) continue;   // idle tables are not persisted
+    out.push([at[0], at[1], at[2], st.recipeId, ...pairs]);
+  }
+  return out;
+}
+
+export function deserializeCraftingTables(
+  a: CraftingTuple[] | null | undefined,
+): Map<string, CraftingTableState> {
+  const m = new Map<string, CraftingTableState>();
+  if (!Array.isArray(a)) return m;
+  for (const t of a) {
+    if (!Array.isArray(t) || t.length < 4) continue;
+    const [x, y, z, rid, ...pairs] = t;
+    if (!Number.isFinite(x) || !Number.isFinite(y) || !Number.isFinite(z)) continue;
+    const buffered: Record<number, number> = {};
+    for (let i = 0; i + 1 < pairs.length; i += 2) {
+      const id = Math.floor(Number(pairs[i]));
+      const n = Math.floor(Number(pairs[i + 1]));
+      if (id > 0 && n > 0) buffered[id] = (buffered[id] ?? 0) + n;
+    }
+    // sanitize: unknown recipe -> null, non-ingredients dropped, counts clamped to
+    // the per-craft requirement so a hand-edited save can never over-buffer.
+    const st = sanitizeCraftingState({ recipeId: typeof rid === 'string' ? rid : null, buffered });
+    m.set(machineKey(x, y, z), st);
+  }
+  return m;
+}
```
