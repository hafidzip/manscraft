  const seed = hashPlant(wx, y, wz, 101);
  const span = Math.max(0, g.blades[1] - g.blades[0]);
  const nb = g.blades[0] + Math.floor(seed * (span + 1));

  for (let i = 0; i < nb; i++) {
    const r = (k: number) => hashPlant(wx, y, wz, 200 + i * 11 + k);
    const cx = x + 0.5 + (r(0) - 0.5) * g.spread;
    const cz = z + 0.5 + (r(1) - 0.5) * g.spread;
    const ang = r(2) * Math.PI * 2;
    const dirX = Math.cos(ang), dirZ = Math.sin(ang);
    const h  = g.height[0] + r(3) * (g.height[1] - g.height[0]);
    const hw = (g.width[0] + r(4) * (g.width[1] - g.width[0])) * 0.5;

    const cAmt  = g.curve * (0.6 + r(5) * 0.8);
    const leanA = (r(6) - 0.5) * 2 * g.lean;
    const phase = r(7) * Math.PI * 2;
    // width axis (horizontal, perpendicular to the blade's facing direction)
    const pX = -dirZ, pZ = dirX;
    // soft mostly-up normal: reads as flat-lit foliage, matching the original tufts
    const nl = Math.hypot(dirX * 0.35, 0.9, dirZ * 0.35);
    const nx = (dirX * 0.35) / nl, ny = 0.9 / nl, nz = (dirZ * 0.35) / nl;

    const segs = g.segments;
    const at = (t: number) => ({
      px: cx + dirX * cAmt * t * t + pX * leanA * t,
      py: y + h * t,
      pz: cz + dirZ * cAmt * t * t + pZ * leanA * t,
      w: hw * (1 - g.taper * t),
    });

    for (let s = 0; s < segs; s++) {
      const t0 = s / segs, t1 = (s + 1) / segs;
      const a = at(t0), b = at(t1);
      const vA = v0 + (v1 - v0) * t0, vB = v0 + (v1 - v0) * t1;
      pushQuad(bucket,
        [a.px - pX * a.w, a.py, a.pz - pZ * a.w],
        [a.px + pX * a.w, a.py, a.pz + pZ * a.w],
        [b.px + pX * b.w, b.py, b.pz + pZ * b.w],
        [b.px - pX * b.w, b.py, b.pz - pZ * b.w],
        [nx, ny, nz],
        [[u0, vA], [u1, vA], [u1, vB], [u0, vB]],
        mul,
        // sway: (phase, strength ∝ height along blade, isTip, unused)
        [[phase, t0, 0, 0], [phase, t0, 0, 0],
         [phase, t1, t1 >= 1 ? 1 : 0, 0], [phase, t1, t1 >= 1 ? 1 : 0, 0]],
      );
    }
  }
}
```

**Flowers** — N cross quads instead of a fixed X:

```ts
/**
 * Cross-quad plant. `quads` controls petal count (task 3):
 *   2 → classic 4-petal X (terran/desert)
 *   3 → 6-petal alien asterisk
 *   4 → 8-petal neon starburst
 * `f === null` reproduces the exact legacy 2-quad behaviour.
 */
function emitCross(
  bucket: Bucket, x: number, y: number, z: number, tile: number,
  swayStrength = 0, wx = x, wz = z,
  f: FlowerShape | null = null, mul: [number, number, number] = [1, 1, 1],
): void {
  const [u0, v0, u1, v1] = tileUV(tile);
  const quads  = f?.quads ?? 2;
  const radius = f?.radius ?? 0.35;   // 0.35 ≡ the old m = 0.15 margin
  const height = f?.height ?? 1.0;
  const twist  = f?.twist ?? 0;
  const phase  = hashPlant(wx, y, wz, 11) * Math.PI * 2;
  const base   = hashPlant(wx, y, wz, 13) * Math.PI * 2 * twist;
  const cx = x + 0.5, cz = z + 0.5;

  for (let q = 0; q < quads; q++) {
    const a = base + (Math.PI * q) / quads;
    const dx = Math.cos(a) * radius, dz = Math.sin(a) * radius;
    const nx = -Math.sin(a), nz = Math.cos(a);
    pushQuad(bucket,
      [cx - dx, y,          cz - dz],
      [cx + dx, y,          cz + dz],
      [cx + dx, y + height, cz + dz],
      [cx - dx, y + height, cz - dz],
      [nx, 0, nz],
      [[u0, v0], [u1, v0], [u1, v1], [u0, v1]],
      mul,
      [[phase, 0, 0, 0], [phase, 0, 0, 0],
       [phase, swayStrength, 1, 0], [phase, swayStrength, 1, 0]],
    );
  }
}
