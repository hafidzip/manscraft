      this.view.poke(x, y, z, id);
      this.onBlock(x, y, z, this.view.getBlock(x, y, z), id);
    }
  }

  /** Verbatim port of engine.laserMinerCanMine (§4.9). */
  private mineable(id: number): boolean {
    if (id === B.AIR || id < 0) return false;
    if (id === B.WATER) return false;
    if (isConveyor(id) || isInserter(id) || isLaserMiner(id) || id === B.TURRET) return false;
    const d = DEFS[id];
    if (!d || !d.solid || !isFinite(d.hardness)) return false;
    return TO_FPS[id] !== undefined;
  }

  private settleCell(x: number, y: number, z: number): number {
    let yy = clampCellY(y);
    for (let i = 0; i < 5; i++) {
      if (yy <= 1) break;
      const below = this.view.getBlock(x, yy - 1, z);
      if (below !== B.AIR && below !== B.WATER) break;
      yy--;
    }
    return ledgerCell(x, yy, z);
  }

  /* ---- laser miners --------------------------------------------------- */
  private acquire(m: SimMachine): boolean {
    const px = -m.dz, pz = m.dx;   // perpendicular
    let best = Infinity, bx = 0, by = 0, bz = 0;
    for (let step = 1; step <= LASER_RANGE; step++) {
      for (let lat = -2; lat <= 2; lat++) {
        for (let vert = -2; vert <= 2; vert++) {
          const x = wrapBlock(m.x + m.dx * step + px * lat);
          const y = m.y + vert;
          const z = wrapBlock(m.z + m.dz * step + pz * lat);
          if (y < 1 || y >= H) continue;
          const score = step * 10 + Math.abs(lat) * 2 + Math.abs(vert) * 2;
          if (score >= best) continue;
          const id = this.view.getBlock(x, y, z);
          if (!this.mineable(id)) continue;
          if (!this.reachable(m, x, y, z, step)) continue;
          best = score; bx = x; by = y; bz = z;
        }
      }
    }
    if (best === Infinity) { m.tx = -1; return false; }
    m.tx = bx; m.ty = by; m.tz = bz; m.charge = 0; m.aim = 0;
    return true;
  }

  /** Coarse LOS: march the axis-major path, allow air/water/ghost machines. */
  private reachable(m: SimMachine, tx: number, ty: number, tz: number, steps: number): boolean {
    for (let s = 1; s < steps; s++) {
      const f = s / steps;
      const x = wrapBlock(Math.round(m.x + (tx - m.x) * f));
      const y = Math.round(m.y + (ty - m.y) * f);
      const z = wrapBlock(Math.round(m.z + (tz - m.z) * f));
      if (x === tx && y === ty && z === tz) break;
      const id = this.view.getBlock(x, y, z);
      if (id === B.AIR || id === B.WATER) continue;
      if ((kindOf(id) & (MK_INSERTER | MK_LASER | MK_TURRET)) !== 0) continue;
      if (DEFS[id]?.solid) return false;
    }
    return true;
  }

  private stepMiners(dt: number): void {
    for (const m of this.machines.values()) {
      if ((m.kind & MK_LASER) === 0) continue;
      let remaining = dt;
      let guard = 8;                        // bound work inside one catch-up quantum
      while (remaining > 0 && guard-- > 0) {
        if (m.tx < 0) { if (!this.acquire(m)) { m.aim = LASER_RETARGET; break; } }
        if (m.aim > 0) { const a = Math.min(m.aim, remaining); m.aim -= a; remaining -= a; continue; }
        const id = this.view.getBlock(m.tx, m.ty, m.tz);
        if (!this.mineable(id)) { m.tx = -1; m.charge = 0; m.aim = LASER_RETARGET; continue; }
        const need = (1 - m.charge) * LASER_MINE_TIME;
        if (remaining < need) { m.charge += remaining / LASER_MINE_TIME; remaining = 0; break; }
        remaining -= need;
        m.charge -= 1; if (m.charge < 0) m.charge = 0;   // carry remainder, never reset
        this.setBlockAuthoritative(m.tx, m.ty, m.tz, B.AIR);
        const fps = TO_FPS[id];
        if (fps !== undefined) {
          // drop at the back of the miner, y+1.1 (§4.9), then settle
          this.ledger.addAtCell(this.settleCell(wrapBlock(m.x - m.dx), m.y + 1, wrapBlock(m.z - m.dz)), fps, 1);
          this.stats.produced++;
        }
        this.stats.mined++;
        m.tx = -1; m.aim = LASER_RETARGET + LASER_AIM;
      }
    }
  }

  /* ---- inserters ------------------------------------------------------ */
  private stepInserters(dt: number): void {
    for (const m of this.machines.values()) {
      if ((m.kind & MK_INSERTER) === 0) continue;
      let remaining = dt;
      let guard = 8;
      while (remaining > 0 && guard-- > 0) {
        if (m.st === 0) {
          m.t += remaining; remaining = 0;
          if (m.t < INSERTER_IDLE_POLL) break;
          let got = 0;
          for (const dy of INPUT_Y_OFFSETS) {
            const cell = ledgerCell(m.x - m.dx, m.y + dy, m.z - m.dz);
            if (this.ledger.takeAnyFromCell(cell, 1) > 0) { got = this.ledger.out.id; break; }
          }
          if (got === 0) { m.t = INSERTER_IDLE_POLL; break; }     // keep polling
          m.held = got; m.st = 1; m.t = 0;
          remaining = 0; // pickup consumed this slice; swing starts next iteration/step
        } else {
          const need = INSERTER_CYCLE - m.t;
          if (remaining < need) { m.t += remaining; remaining = 0; break; }
          remaining -= need;
          this.release(m);
          m.st = 0; m.t = 0;                                       // remainder rolls forward
        }
      }
    }
  }

  private release(m: SimMachine): void {
    if (m.held <= 0) return;
    const fx = wrapBlock(m.x + m.dx), fz = wrapBlock(m.z + m.dz);
    if (FEED_FURNACES) {
      const fid = this.view.getBlock(fx, m.y, fz);
      if (fid === B.FURNACE || fid === B.FURNACE_LIT) { /* slot insertion — disabled, see flag */ }
    }
    this.ledger.addAtCell(this.settleCell(fx, m.y + 1, fz), m.held, 1); // release height y+1.22
    m.held = 0;
    this.stats.inserted++;
  }

  /* ---- furnaces -------------------------------------------------------- */
  private stepFurnaces(dt: number): void {
    for (const [k, st] of this.furnaces) {
      const wasLit = st.burn > 0;
      const changed = tickFurnace(st, dt);
      if (this.live && changed) {
        const isLit = st.burn > 0;
        if (isLit !== wasLit) {
          const p = k.split(',');
          const x = +p[0], y = +p[1], z = +p[2];
          const cur = this.live.peekBlock(x, y, z);
          if (cur === B.FURNACE || cur === B.FURNACE_LIT)
            this.live.setBlock(x, y, z, isLit ? B.FURNACE_LIT : B.FURNACE);
        }
      }
      if (changed) this.stats.smelted++;
      if (furnaceIdle(st) && !this.pinnedFurnaces.has(k)) this.furnaces.delete(k);
    }
  }
  /** Engine pins the open/UI furnace so it is not GC'd mid-interaction. */
  readonly pinnedFurnaces = new Set<string>();

  furnaceAt(x: number, y: number, z: number, create = false): FurnaceState | null {
    const k = furnaceKey(x, y, z);
    let st = this.furnaces.get(k) ?? null;
    if (!st && create) { st = newFurnace(); this.furnaces.set(k, st); }
    return st;
  }

  /* ---- clock ----------------------------------------------------------- */
  private step(dt: number): void {
    if (this.beltsDirty) { this.belts.rebuild({ forEachConveyor: this.forEachConveyorCb }, this.view); this.beltsDirty = false; }
    this.stepMiners(dt);
    this.belts.step(dt, this.ledger, this.view);
    this.stepInserters(dt);
    this.stepFurnaces(dt);
    this.simTimeSec += dt;
    this.stats.steps++;
  }

  /** Queue wall time without simulating (cheap; used by the background clock). */
  enqueue(sec: number): void {
    if (sec <= 0) return;
    this.backlogSec = Math.min(this.backlogSec + sec, MAX_BACKLOG);
    this.lastWallMs = Date.now();
  }

  /**
   * Burn backlog in ≤ MAX_STEP quanta, resumable under a millisecond budget
   * (same pattern as World.prepareAllData). Returns simulated seconds.
   */
  advance(elapsedSec = 0, budgetMs = Infinity): number {
    this.enqueue(elapsedSec);
    const t0 = now();
    let done = 0;
    while (this.backlogSec > 1e-4) {
      const step = Math.min(this.backlogSec, MAX_STEP);
      this.step(step);
      this.backlogSec -= step; done += step;
      if (budgetMs !== Infinity && now() - t0 >= budgetMs) break;
    }
    this.stats.catchUpSec += done;
    return done;
  }
