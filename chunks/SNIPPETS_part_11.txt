```ts
// src/game/render/planetLook.ts
import * as THREE from 'three';
import type { PlanetTheme } from '../space/theme';
import { colorsFor } from '../world/blocks';
import { NO_ORIGIN, originGroupHex, type OriginTag } from '../core/origin';
import type { SlotItem } from '../fps/Inventory';

export interface PlanetLook {
  sky: THREE.Color; fog: THREE.Color; fogDensity: number;
  ambient: THREE.Color; sun: THREE.Color;
  particle: THREE.Color; toolTint: THREE.Color;
  water: THREE.Color; underwater: THREE.Color;
}

const C = (hex: number) => new THREE.Color(hex);

export function planetLook(theme: PlanetTheme | null): PlanetLook {
  if (!theme) {
    return { sky: C(0x87ceeb), fog: C(0xa8c8e8), fogDensity: 0.0075, ambient: C(0xffffff),
             sun: C(0xfff6e6), particle: C(0x795a3a), toolTint: C(0xffffff),
             water: C(0x3a66de), underwater: C(0x1b3a7a) };
  }
  const water = C(theme.waterHex);
  return {
    sky: C(theme.skyHex),
    fog: C(theme.fogHex),
    fogDensity: theme.fogDensity,
    ambient: C(theme.ambientHex),
    sun: C(theme.sunHex),
    particle: C(theme.particleHex),
    toolTint: C(theme.toolTintHex),
    water,
    underwater: water.clone().multiplyScalar(0.55),
  };
}

/** Called once per planet load, right after setActivePlanetTheme. */
export function applyPlanetLook(
  look: PlanetLook,
  r: {
    scene: THREE.Scene; renderer: THREE.WebGLRenderer;
    ambient?: THREE.AmbientLight | THREE.HemisphereLight; sun?: THREE.DirectionalLight;
    waterMat?: THREE.Material & { color?: THREE.Color };
    skyMat?: THREE.Material & { color?: THREE.Color; uniforms?: Record<string, { value: unknown }> };
  },
): void {
  r.scene.fog = new THREE.FogExp2(look.fog.getHex(), look.fogDensity);
  r.scene.background = look.sky;
  r.renderer.setClearColor(look.sky);
  if (r.ambient) r.ambient.color.copy(look.ambient);
  if (r.sun) r.sun.color.copy(look.sun);
  if (r.waterMat?.color) r.waterMat.color.copy(look.water);
  if (r.skyMat?.uniforms?.topColor) (r.skyMat.uniforms.topColor.value as THREE.Color).copy(look.sky);
  else if (r.skyMat?.color) r.skyMat.color.copy(look.sky);
}

/**
 * Held item / tool tint (task 6 + 7): a held block shows ITS OWN planet's colour, so a
 * jungle log in your hand still looks jungle while you stand on an ice world.
 */
export function heldTint(item: SlotItem | null, fallback: PlanetLook): THREE.Color {
  if (!item || item.kind !== 'block') return fallback.toolTint.clone();
  const tag: OriginTag = item.origin ?? NO_ORIGIN;
  const cols = colorsFor(item.blockId, tag);
  return cols.length ? new THREE.Color(cols[0]) : fallback.toolTint.clone();
}

/** Break/step particle colour for a specific voxel, from ITS origin, not the planet's. */
export function particleColorFor(blockId: number, tag: OriginTag, fallback: PlanetLook): THREE.Color {
  const cols = colorsFor(blockId, tag);
  return cols.length ? new THREE.Color(cols[(cols.length * 0.5) | 0]) : fallback.particle.clone();
}

/** Convenience for UI swatches (inventory slot border / hotbar glow). */
export const uiSwatch = (tag: OriginTag): number => originGroupHex(tag, 'grass');
```
