
export class ShadowLod {
  private rt: THREE.WebGLRenderTarget;
  private cam: THREE.OrthographicCamera;
  private depthMat: THREE.MeshDepthMaterial;
  private lastFar = new THREE.Vector3(Infinity, 0, Infinity);
  private clear = new THREE.Color();
  private nearCell = new THREE.Vector2(NaN, NaN);
  private prevNearCell = new THREE.Vector2(NaN, NaN);
  dirty = true;                       // geometry changed -> both cascades stale

  constructor(atlas: THREE.Texture) {
    this.rt = new THREE.WebGLRenderTarget(SHADOW_FAR_SIZE, SHADOW_FAR_SIZE, {
      minFilter: THREE.NearestFilter, magFilter: THREE.NearestFilter,
      format: THREE.RGBAFormat, type: THREE.UnsignedByteType,
      depthBuffer: true, stencilBuffer: false, generateMipmaps: false,
    });
    this.rt.texture.name = 'shadowFar';

    this.cam = new THREE.OrthographicCamera(
      -SHADOW_FAR_HALF_EXTENT, SHADOW_FAR_HALF_EXTENT,
      SHADOW_FAR_HALF_EXTENT, -SHADOW_FAR_HALF_EXTENT,
      SHADOW_FAR_NEAR, SHADOW_FAR_FAR,
    );
    // NOTE: matrixAutoUpdate MUST stay true — `snapShadowCamera` calls
    // updateMatrixWorld(true), which only recomposes the local matrix when
    // matrixAutoUpdate is set. (Silent, ugly bug if you "optimise" this.)
    //
    // Layer filter = exact caster control. `WebGLShadowMap.renderObject` tests
    // object.layers against the VIEW camera, so three's own cascade is
    // unaffected by this bit; only THIS pass sees the filter.
    this.cam.layers.set(LAYER_FAR_CASTER);

    // ONE override material for the whole pass. The atlas supplies the alpha
    // test that leaves need, and every solid block face in the atlas is
    // alpha = 1, so terrain is unaffected. Cheaper than per-mesh
    // customDepthMaterial dispatch, and it cannot be defeated by a chunk that
    // forgot to set one.
    this.depthMat = new THREE.MeshDepthMaterial({
      depthPacking: THREE.RGBADepthPacking,
      map: atlas, alphaTest: 0.4, side: THREE.DoubleSide,
    });

    SHADOW_UNIFORMS.uFarShadowMap.value = this.rt.texture;
    // Ortho depth is LINEAR, so a world-space bias converts exactly. Never
    // hand-tune a normalised bias — it silently changes when near/far change.
    SHADOW_UNIFORMS.uFarShadowBias.value =
      -SHADOW_FAR_BIAS_METRES / (SHADOW_FAR_FAR - SHADOW_FAR_NEAR);
    // Nothing has been rendered yet. An unbound sampler reads as black, which
    // unpacks to depth 0 = "everything is shadowed". Stay inert until the
    // first pass lands.
    SHADOW_UNIFORMS.uFarShadowStrength.value = 0;
  }

  /** Geometry changed (chunk (re)meshed, block placed/broken) → both cascades stale. */
  markDirty(): void {
    this.dirty = true;
  }

  /** Global off-switch for weak GPUs. 0 = far cascade costs one branch per fragment. */
  setStrength(v: number): void {
    SHADOW_UNIFORMS.uFarShadowStrength.value = v;
  }

  // ---------------------------------------------------------------------------
  // NEAR cascade — three renders it; we only own the frustum placement.
  // Call this ONLY on the frame you are about to set `needsUpdate`, never every
  // frame: a frustum that does not move cannot swim, and one that moves by a
  // whole number of texels cannot shimmer. Those are the only two states.
  // ---------------------------------------------------------------------------
  syncNear(light: THREE.DirectionalLight, dir: THREE.Vector3, centre: THREE.Vector3): void {
    snapShadowLight(
      light, dir, centre,
      CELESTIAL_SHADOW_HALF_EXTENT, light.shadow.mapSize.x,
      CELESTIAL_SHADOW_DISTANCE, this.nearCell,
    );
    light.shadow.needsUpdate = true;   // per-light gate (shadow.autoUpdate = false)
  }

  /** Did the last `syncNear` actually land on a new texel cell? (debug/telemetry) */
  nearMoved(): boolean {
    const moved = !this.nearCell.equals(this.prevNearCell);
    this.prevNearCell.copy(this.nearCell);
    return moved;
  }

  // ---------------------------------------------------------------------------
  // FAR cascade — our own pass, cached aggressively. This is where the static
  // sun pays for itself: a constant direction + an 8 m recentre cell means the
  // far map is typically valid for 10-40 seconds of walking.
  // ---------------------------------------------------------------------------
  private needsFar(dir: THREE.Vector3, centre: THREE.Vector3): boolean {
    if (this.dirty) return true;
    if (!this.lastDir.equals(dir)) return true;            // sun <-> moon swap
    const dx = centre.x - this.lastFar.x;
    const dz = centre.z - this.lastFar.z;
    if (dx * dx + dz * dz > SHADOW_FAR_RECENTER * SHADOW_FAR_RECENTER) return true;
    // Vertical motion matters far less (the frustum is 192 m tall), but flying
    // straight up must still recentre eventually.
    return Math.abs(centre.y - this.lastFar.y) > SHADOW_FAR_RECENTER * 4;
  }

  /**
   * MUST be called during the update phase, BEFORE the main render — it binds
   * its own render target and would otherwise stomp the post-processing chain.
   * @returns true if the far map was re-rendered this frame.
   */
  update(
    renderer: THREE.WebGLRenderer, scene: THREE.Scene,
    dir: THREE.Vector3, centre: THREE.Vector3,
  ): boolean {
    if (!this.needsFar(dir, centre)) return false;

    snapShadowCamera(
      this.cam, dir, centre,
      SHADOW_FAR_HALF_EXTENT, SHADOW_FAR_SIZE, SHADOW_FAR_DISTANCE,
    );
    SHADOW_UNIFORMS.uFarShadowMatrix.value
      .copy(BIAS)
      .multiply(this.cam.projectionMatrix)
      .multiply(this.cam.matrixWorldInverse);   // Camera.updateMatrixWorld refreshed it

    // ---- save every piece of renderer/scene state we touch ------------------
    const prevTarget = renderer.getRenderTarget();
    const prevBg = scene.background;
    const prevOverride = scene.overrideMaterial;
    const prevAutoClear = renderer.autoClear;
    const prevShadowNeeds = renderer.shadowMap.needsUpdate;
    const prevAlpha = renderer.getClearAlpha();
    renderer.getClearColor(this.clear);

    // `scene.background` is a live THREE.Color (Sky sets it), so three would
    // clear this depth map to the SKY COLOUR. Depth-packed maps must clear to
    // white = "infinitely far", exactly like three's own shadow pass does.
    scene.background = null;
    scene.overrideMaterial = this.depthMat;
    // Do not let this extra render() consume or trigger the near cascade.
    renderer.shadowMap.needsUpdate = false;
    renderer.autoClear = true;
    renderer.setClearColor(0xffffff, 1);
    renderer.setRenderTarget(this.rt);
    renderer.render(scene, this.cam);

    // ---- restore -----------------------------------------------------------
    renderer.setRenderTarget(prevTarget);
    renderer.setClearColor(this.clear, prevAlpha);
    renderer.autoClear = prevAutoClear;
    renderer.shadowMap.needsUpdate = prevShadowNeeds;
    scene.overrideMaterial = prevOverride;
    scene.background = prevBg;

    this.lastFar.copy(centre);
    this.lastDir.copy(dir);
    this.dirty = false;
    SHADOW_UNIFORMS.uFarShadowStrength.value = 1;
    return true;
  }

  dispose(): void {
    this.rt.dispose();
    this.depthMat.dispose();
  }
}
