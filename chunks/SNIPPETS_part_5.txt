+      ? { kind: 'block', blockId, count, origin: tag }
+      : { kind: 'block', blockId, count };
+    return stackItem(this.hotbar, this.mainInv, item);
+  }
+
+  /** Origin of the currently-selected hotbar block, for placement + held-item tinting. */
+  selectedOrigin(): OriginTag {
+    const it = this.getItem?.(this.selected ?? 0) ?? null;
+    return it && it.kind === 'block' ? (it.origin ?? NO_ORIGIN) : NO_ORIGIN;
+  }
```

`swapSlots` / `consumeAt` / `setItem` / `getItem` move whole `SlotItem` objects, so
`origin` rides along with **zero** changes — that's why the optional-field approach was
chosen. `countBlock(id)` / `canAdd(id)` intentionally stay **origin-agnostic** (you have
"12 logs" regardless of provenance) which is exactly what shared recipes need.

  dirt: WB.DIRT, grass: WB.GRASS, snow: WB.SNOW, glass: WB.GLASS, coal: WB.COAL_ITEM,
  stick: WB.STICK_ITEM, cactus: WB.CACTUS,
};
/** Non-categorised items that still need to cross the boundary. */
const EXPLICIT: Record<IdSpace, Record<number, number>> = {
