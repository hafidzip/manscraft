+    const recipe = input ? smeltResult(input.blockId) : null;
+    const canCook = !!recipe;
+
+    if (st.burn <= 0 && canCook && st.fuel && st.fuel.kind === 'block') {
+      const t = fuelTime(st.fuel.blockId);
+      if (t > 0) {
+        st.burn = t;
+        st.burnMax = t;
+        st.fuel.count -= 1;
+        if (st.fuel.count <= 0) st.fuel = null;
+        changed = true;
+      }
+    }
+
+    if (st.burn <= 0) {                        // unlit: cook progress decays, tick ends
+      if (st.cook > 0) { st.cook = Math.max(0, st.cook - remaining * 2); changed = true; }
+      break;
+    }
+
+    if (!canCook) {                            // lit with nothing to smelt: burn down
+      const step = Math.min(remaining, st.burn);
+      st.burn -= step;
+      remaining -= step;
+      if (st.cook > 0) st.cook = Math.max(0, st.cook - step * 2);
+      changed = true;
+      continue;
+    }
+
+    const step = Math.min(remaining, st.burn, SMELT_TIME - st.cook);
+    st.burn = Math.max(0, st.burn - step);
+    st.cook += step;
+    remaining -= step;
+    changed = true;
+
+    if (st.cook >= SMELT_TIME - 1e-9 && recipe && input) {
+      // One completed smelt. Never more than one item type reported per tick.
+      if (out && out.producedId !== 0 && out.producedId !== recipe.output) break;
+      st.cook = 0;
+      input.count -= 1;
+      if (input.count <= 0) st.input = null;
+      if (out) {
+        out.producedId = recipe.output;
+        out.producedCount += recipe.count;
+      } else if (st.output && st.output.kind === 'block' && st.output.blockId === recipe.output) {
+        st.output.count += recipe.count;
+      } else {
+        st.output = { kind: 'block', blockId: recipe.output, count: recipe.count };
+      }
+    }
+  }
+
+  const litChanged = wasLit !== st.burn > 0;
+  if (out) out.changed = changed || litChanged;
+  return changed || litChanged;
+}
```

`outputAccepts()` becomes dead for the ejecting path — keep it only if the manual UI still reads `st.output`; otherwise delete it. `furnaceIdle()` is unchanged and now reliably returns true once a furnace drains, so idle furnaces still get garbage-collected.
