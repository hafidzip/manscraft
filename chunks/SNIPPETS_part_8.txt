    } else {
      for (const t of this.hooks.targets()) this.collect(t);   // exact old fallback
    }

    const n = candD.length;
    let rays = 0;
    let best: TurretTarget | null = null;

    // Selection instead of sort: at most MAX_RAYS_PER_ACQUIRE passes of O(n) integer
    // compares. No sort, no comparator closure, no allocation.
    while (rays < MAX_RAYS_PER_ACQUIRE) {
      let m = -1;
      let md = Infinity;
      for (let i = 0; i < n; i++) {
        const d = candD[i];
        if (d < md) { md = d; m = i; }
      }
      if (m < 0) break;                              // all candidates consumed
      candD[m] = Infinity;                           // mark tried
      const t = cand[m];
      if (!t.alive) continue;

      const ex = this.cx0 + minImageF(t.pos.x - this.cx0);
      const ez = this.cz0 + minImageF(t.pos.z - this.cz0);
      tmpDir.set(ex - eye.x, t.pos.y + 1.0 - eye.y, ez - eye.z)
        .divideScalar(md || 1);

      rays++; this.stats.rays++;
      const hit = this.hooks.raycast(eye, tmpDir, md);
      if (hit && hit.dist < md - 0.4) continue;      // occluded: identical predicate
      best = t;
      break;
    }

    g.target = best;
  }
}

/** Module scratch, unchanged in spirit: one eye vector and one direction vector. */
const tmpEye = new THREE.Vector3();
const tmpDir = new THREE.Vector3();
```

**Behavior deltas, stated precisely.**

| Aspect | Old | New |
|---|---|---|
| Target chosen | nearest alive non-peaceful enemy within 22 with a clear ray | identical, unless the 4 nearest in-range enemies are *all* occluded |
| Raycasts per acquire | up to (enemies within range) | ≤ `MAX_RAYS_PER_ACQUIRE` (4) |
| Enemy list walks | guns × enemies, every 0.18 s | enemies once per frame (rebuild) + enemies-near-gun per acquire |
| Cadence | 0.18 s | 0.18 s |
| 360° rotation, fire, damage, fx | — | untouched |

Set `MAX_RAYS_PER_ACQUIRE = Number.POSITIVE_INFINITY` for bit-exact parity; the index
change alone removes the O(guns × all enemies) term, which is the actual spike source.
The bound only matters in a pile-up of ≥4 occluded enemies closer than a visible one,
where the gun idles for one extra 0.18 s slice.

### 3.4 Agent contract addendum

Add one optional member to `MachineAgent` in §3.0. The scheduler uses it to avoid
destroying an entity that holds transient gameplay state:

```ts
// src/game/fps/machineScheduler.ts  (part 1, types) — ADD to interface MachineAgent
  /** Return true to veto distance-based reclamation this frame (e.g. mid-carry). */

```

```ts
// src/game/fps/Inserter.ts — ADD to InserterManager
  retain(key: number): boolean {
    const arm = this.arms.get(key);
    return !!arm && arm.state !== 'idle';   // never reclaim an arm carrying an item
  }
```

`LaserMinerManager` and `TurretManager` do not implement `retain`: all of their state
(`target`, `charge`, `acqT`) is re-derivable from the record and the world, so a
rebuild on approach is indistinguishable from a retained entity.

**Y band semantics.** `yLo`/`yHi` are *relative to* `Math.floor(player.y)`, exactly
reproducing the historic loops: inserters `cy-2 .. cy+3`, laser miners and turrets
`cy-4 .. cy+4`. The scheduler converts them to absolute bounds once per frame and
clamps to `[0, WORLD_HEIGHT-1]`.
