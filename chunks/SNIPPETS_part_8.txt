export const DESPAWN_R       = 136;   // cull wild agents past this, into the pool
export const SPAWN_RATE_MIN  = 6;     // hostiles/s when the crowd is full
export const SPAWN_RATE_MAX  = 60;    // hostiles/s when the crowd is empty
export const SPAWN_PER_FRAME = 6;     // hard ceiling on constructions per frame
export const SPAWN_BUDGET_MS = 1.2;   // …and a wall-clock ceiling on top
export const POOL_MAX        = 256;   // banked instances per preset

export const RELOCATE_R_SQ = RELOCATE_R * RELOCATE_R;
export const DESPAWN_R_SQ  = DESPAWN_R  * DESPAWN_R;
```

**The radius chain is the correctness argument. It must hold, in this order:**

```
RENDER_RADIUS(80) < SPAWN_R_FAR(88) < RELOCATE_R(96) < SIM_RADIUS(112) < DESPAWN_R(136)
```

- `RENDER_RADIUS < SPAWN_R_FAR` → nothing ever spawns inside the scene graph, so no pop-in.
- `RENDER_RADIUS < RELOCATE_R` → a relocated agent is provably detached (Phase 1) at the instant it teleports. **No player can ever witness a relocation.** This is the invariant the original lacked.
- `SPAWN_R_FAR < RELOCATE_R` → an agent can't be spawned and immediately re-relocated (thrash).
- `SIM_RADIUS < DESPAWN_R` → despawn never removes an agent that is still simulating.

recycleTo(x: number, y: number, z: number) {
  this.pos.set(x, y, z);
  this.group.position.set(x, y, z);  // snap, or you get a one-frame streak
  this.vel.set(0, 0, 0);

  this.hp = this.cfg.hp;
  this.tier = 0;
  this.d2 = 0;
  this.coldAccum = 0;
  this.phase = (this.phase * 1664525 + 1013904223) >>> 0;  // re-stagger
  this.home.x = x; this.home.z = z;                        // mutate, don't realloc
  this.invalidatePath();
  this.lastKnown.set(x, y, z);
  this.lastKnownChangeT = 0;
  this.attackCd = 0; this.hurtT = 0; this.deathT = 0;
  this.matrixDirty = true;
  // Phase 1 left matrixAutoUpdate = false while detached. One explicit update,
  // or the agent renders at its previous location for a frame on re-entry.
  this.group.updateMatrixWorld(true);
}

/** Drop every cached route. Safe at any time. */

  this.path = null;
  this.pathIdx = 0;
  this.repathT = 0;
  this.corridorCached = null;
  this.directBlockT = 0;
}

  this.detachFromScene();

  this.corridorCached = null;
}

Stale `lastKnown` is the subtle one: without resetting it, a relocated agent spends its first seconds walking toward where the player *was* relative to its old position — visibly wrong-way movement at the ring.

### `EnemyPool` (new file, ~40 lines)

```ts
// ── src/game/fps/EnemyPool.ts ─────────────────────────────────────────────
import type { Enemy } from './Enemy';

export class EnemyPool {
  private free = new Map<string, Enemy[]>();
  size = 0;
  takes = 0; makes = 0; drops = 0;   // telemetry: makes should flatline after warmup

  take(preset: string): Enemy | null {
    const list = this.free.get(preset);
    const e = list && list.length ? list.pop()! : null;
    if (e) { this.size--; this.takes++; } else { this.makes++; }
    return e;
  }

  give(e: Enemy, max = 256): void {
    e.alive = false;
    e.detachFromScene();             // Phase 1: out of the graph = zero per-frame cost
    let list = this.free.get(e.poolKey);
    if (!list) { list = []; this.free.set(e.poolKey, list); }
    if (list.length >= max) { e.dispose(); this.drops++; return; }
    list.push(e);
    this.size++;
  }
}
```

Pooling is **keyed by preset**. A banked `heavy` cannot be handed out as a `grunt` — the mesh is wrong. The original doc's "no constructor cost" claim only holds with this key.

```ts
/** Replaces the entire body of wildTick. */
private wildTick(dt: number) {
  const deficit = MAX_POP - this.hostiles;
  if (deficit <= 0 || !this.night) { this.spawnCredit = 0; return; }

  // Rate ramps with the deficit: fast to fill an empty world, slow to top up
  // a full one. This is the fix for "87 minutes to reach the cap".
  const fill = deficit / MAX_POP;                        // 1 = empty, 0 = full
  const rate = SPAWN_RATE_MIN + (SPAWN_RATE_MAX - SPAWN_RATE_MIN) * fill;
  this.spawnCredit += rate * dt;
  // Clamp: a 400 ms hitch must not produce a 24-agent construction spike.
  if (this.spawnCredit > SPAWN_PER_FRAME) this.spawnCredit = SPAWN_PER_FRAME;

  let budget = Math.min(this.spawnCredit | 0, SPAWN_PER_FRAME, deficit);
  const tEnd = performance.now() + SPAWN_BUDGET_MS;
  while (budget-- > 0) {
    if (performance.now() > tEnd) break;   // amortise across frames
    if (!this.spawnOneWild()) { this.refusals++; break; }  // no valid site
    this.spawnCredit -= 1;
  }
}
```

```ts
/** Picks a site. Returns true if an agent was placed. */
private spawnOneWild(): boolean {
  const pp = this.player.pos;
  // The ring grows with the crowd. Spawning *into* the pack makes the Phase 3
  // separation solver detonate — two agents at r < 0.001 get no push, and the
  // ones at r = 0.1 get a 20x impulse.
  const grow = this.hostiles / MAX_POP;
  const rMin = SPAWN_R_NEAR + (SPAWN_R_FAR - SPAWN_RING_W - SPAWN_R_NEAR) * grow;
  const rMax = rMin + SPAWN_RING_W;

  for (let tries = 0; tries < 8; tries++) {
    const a = Math.random() * Math.PI * 2;
    const r = rMin + Math.random() * (rMax - rMin);
    const fx = wrapBlock(Math.floor(pp.x + Math.cos(a) * r));
    const fz = wrapBlock(Math.floor(pp.z + Math.sin(a) * r));

    const h = this.deps.world.highestY(fx, fz);
    if (h < 2 || h >= WORLD_HEIGHT - 3) continue;
    const floor = this.deps.world.get(fx, h - 1, fz);
    if (isWaterId(floor) || floor === B.LEAVES || floor === B.AIR) continue;
    if (this.deps.world.solid(fx, h, fz) || this.deps.world.solid(fx, h + 1, fz)) continue;

    const sx = fx + 0.5, sz = fz + 0.5;

    // Reuse the Phase 3 grid to reject occupied cells. Free — the grid is
    // already built this frame.
    let blocked = false;
    this.grid.forEachNear(sx, sz, 1.2, (b) => {
      if (blocked || !b.alive) return;
      const bdx = wrapDelta(b.pos.x - sx, WORLD_SIZE);
      const bdz = wrapDelta(b.pos.z - sz, WORLD_SIZE);
      if (bdx * bdx + bdz * bdz < 1.0) blocked = true;
    });
