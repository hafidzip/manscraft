  }
}

### `src/game/world/mesher.ts` (tall grass emission — blank if tile empty)

```ts
// In buildChunkGeometry:
const grassShape = activeFlora().grass;
const flowerShape = activeFlora().flower;

if (d.cross) {
  const plantTag = tagAt(x, y, z);
  const plantMul = relativeColorMul(plantTag, nativeTag, BLOCK_GROUP[id]);
  if (id === B.TALLGRASS) {
    if (grassShape.present) emitTallGrass(cutoutB, x, y, z, d.side, baseX + x, baseZ + z, plantMul);
  } else if (id === B.FLOWER_RED || id === B.FLOWER_YELLOW) {
    if (flowerShape.present) emitCross(cutoutB, x, y, z, d.side, 0.018, baseX + x, baseZ + z, plantMul);
  } else {
    emitCross(cutoutB, x, y, z, d.side, 0.028, baseX + x, baseZ + z, plantMul);
  }
  continue;
}

// Opaque faces use:
// const tile = f.dir[1] === 1 ? d.top : f.dir[1] === -1 ? d.bottom : d.side;
// const [u0, v0, u1, v1] = tileUV(tile);

function emitTallGrass(
  bucket: Bucket, x: number, y: number, z: number, tile: number, wx: number, wz: number,
  mul: [number, number, number] = [1, 1, 1],
): void {
  const [u0, v0, u1, v1] = tileUV(tile);
  const seed = hashPlant(wx, y, wz, 101);
  const blades = 5 + Math.floor(seed * 3);
  for (let i = 0; i < blades; i++) {
    // ... builds thin quads sampling FULL tile UV (u0..u1, v0..v1)
    // normals encode grass: bucket.nrm.push(cx, cz, ang + 100.0);
    bucket.uv.push(u0, v0, u0, v1, u1, v1, u1, v0);
    // vertex colors multiply texture — keep mul near 1,1,1 for native grass
  }
}
```

**Important:** tallgrass quads sample the **entire** 16×16 tile UV. If the painter only draws a few 1px blades, most of the quad is transparent (OK with alphaTest), but if the painter draws nothing → **blank grass**. Prefer denser blade pixels and/or a faint filled silhouette so alphaTest retains visible fragments. Optionally improve mesher later to use tighter UVs; texture fix is the first line of defense.

### `src/game/core/noise.ts`

```ts
export function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
```

1. **Add `PlanetPaintContext`** in `textures.ts`:
   - `type`, `seedU32`, `originTag`, hex→RGB for grass/dirt/stone/sand/log/leaf/water from `theme.colors` or `groupTargetRgb` + hueBucket nudge.
   - motif style enum derived from type: `lush | dry | frozen | ash | alien | fungal | crimson | ...`.

2. **Change painter signature** to `(img, r, ctx) => void` OR close over `ctx` when building a `PAINTERS` map inside `createTextures(theme)`.

3. **Rewrite organic painters** with Minecraft pixel-art techniques using `ctx` palettes:
   - `grass_top`, `grass_side`, `dirt`, `stone`, `sand`, `leaves`, `log_side`, `log_top`, `tallgrass`, `flower_red`, `flower_yellow`, `snow*`, `cactus*`, `water`, `planks` (wood matches log).
   - Keep machines/ores mostly as-is (optional subtle stone base match).

4. **Seed RNG** from planet:
   ```ts
   const seedU32 = Number(theme.seed & 0xffffffffn) ^ (theme.originTag << 16);
   painter(img, mulberry32(seedU32 ^ (i * 7919) ^ 0x9e3779b9));
   ```

   - Flower painters always have stem + petals with a>0.
