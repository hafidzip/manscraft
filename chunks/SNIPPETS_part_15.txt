  }

  private grow(blockId: number, g: Group): void {
    const cap = g.mesh.instanceMatrix.count * 2;
    const next = this.make(blockId, cap);
    (next.mesh.instanceMatrix.array as Float32Array)
      .set(g.mesh.instanceMatrix.array as Float32Array);
    next.count = g.count;
    next.mesh.count = g.count;
    for (let i = 0; i < g.count; i++) next.owners[i] = g.owners[i];
    next.dirty = true;
    this.scene.remove(g.mesh);
    g.mesh.dispose();
    this.groups.set(blockId, next);
    g.mesh = next.mesh; g.owners = next.owners; g.count = next.count; g.dirty = true;
    this.stats.grows++;
  }

  acquire(it: DroppedItem): number {
    const g = this.groupFor(it.blockId);
    if (g.count >= g.mesh.instanceMatrix.count) this.grow(it.blockId, g);
    const slot = g.count++;
    g.owners[slot] = it;
    g.mesh.count = g.count;
    g.dirty = true;
    this.stats.instances++;
    return slot;
  }

  release(it: DroppedItem): void {
    const g = this.groups.get(it.blockId);
    if (!g || it.inst < 0) return;
    const last = --g.count;
    if (it.inst !== last) {
      g.mesh.getMatrixAt(last, this.m);
      g.mesh.setMatrixAt(it.inst, this.m);
      const moved = g.owners[last];
      g.owners[it.inst] = moved;
      if (moved) moved.inst = it.inst;
    }
    g.owners[last] = null;
    g.mesh.count = g.count;
    g.dirty = true;
    this.stats.instances--;
  }

  /** Y-rotation + translation written straight into the matrix: no Euler/Quaternion. */
  set(it: DroppedItem, x: number, y: number, z: number, ry: number): void {
    const g = this.groups.get(it.blockId);
    if (!g || it.inst < 0) return;
    const c = Math.cos(ry), s = Math.sin(ry);
    this.m.set(
      c, 0, s, x,
      0, 1, 0, y,
     -s, 0, c, z,
      0, 0, 0, 1,
    );
    g.mesh.setMatrixAt(it.inst, this.m);
    g.dirty = true;
    this.stats.writes++;
  }

  flush(): void {
    for (const g of this.groups.values()) {
      if (!g.dirty) continue;
      g.mesh.instanceMatrix.needsUpdate = true;
      g.dirty = false;
    }
  }

  dispose(): void {
    for (const g of this.groups.values()) {
      this.scene.remove(g.mesh);
      g.mesh.dispose();
    }
    this.groups.clear();
  }
}
```

Sleeping items are not re-written each frame, so their last matrix stands. That is
consistent with the sleep rule: nothing sleeps within 24 blocks of the player, so a
frozen bob is never observable.

### 5.10 Acceptance checks for S1–S5

Not the full regression suite from item 12, but the minimum that must pass before each
stage lands. All are cheap and mostly headless.

1. **Packing round-trip.** For 100 k random `(x,y,z)` in `[-1024,1535]³`,
   `cellX(packCell(x,y,z)) === wrapBlock(x)`, same for Z, `cellY === y & 127`.
2. **Bucket identity.** `packChunk(cx,cz) === (wrapChunk(cx) << 5) | wrapChunk(cz)` for
   all 32×32 pairs and for negative inputs.
3. **Registry equals brute force.** Place/rotate/remove 500 random machine blocks
   through the normal set-block path, then assert `registry.getAt(x,y,z)` agrees with a
   full `peekBlock` sweep, and that `forEachNear` from 200 random positions returns
   exactly the brute-force filtered set (including from positions on the seam,
   x ∈ {0, 1, 510, 511}).
4. **Chunk reload preserves identity.** Force eviction of a chunk containing an
   inserter, then re-enter: the record's `key` is unchanged, `live` goes false then true,
   and no `onRemove` fires unless the block really disappeared.
5. **Generated chunks skip the rescan.** With `trustGenerator: true`,
   `stats.rescansSkipped` increases and `stats.records` stays constant while streaming
   virgin terrain.
6. **Machine removal is same-frame.** Mine an inserter while it is visible: the arm's
   `THREE.Group` leaves the scene in the same frame (via `onRemove`), matching the old
   `id >= 0 && !isInserter(id)` branch.
7. **Conveyor backpressure.** Belt feeding a laser miner with an inserter in front: the
   lead item stops in the cell *outside* the machine and stays within 0.52 of the
   inserter's pickup point; the inserter drains it; items queue behind without entering
   the ghost cell.
8. **Seam pickup.** Inserter at x = 511 facing +X with an item at x = 0: `takeAt`
   succeeds (this fails before S5a — it is the bug fix in §5.5).
9. **Missing neighbor data.** With the target chunk's data evicted, laser
   `acquire()`/`clearPath()` treat `-1` exactly as before (no target, no beam) and no
   chunk is generated: assert `world.chunks.size` is unchanged across 600 frames.
10. **Sleep/wake.** Settle an item 40 blocks away, confirm it sleeps; place a conveyor
    under it and confirm it wakes within one frame; walk to 20 blocks and confirm it
    wakes within 0.3 s; confirm pickup works on a sleeping item without walking away.
11. **No steady-state garbage.** 60 s of a running factory with the heap profiler: zero
    allocations attributable to `MachineScheduler.update`, `registry.forEachNear`,
    `ItemDropManager.update`, `takeAt`, or `blocked`.
12. **Frame time.** Median and p99 `machines.update` + `itemDrops.update` versus the
    baseline, at 200 machines and 400 items. The 0.5 s scan spikes must disappear from
    the p99 entirely.

## 6. What is still open (deliverable items 6–12)

This revision closes items 1–5. The rest is untouched by S1–S5 and can be merged
forward independently; nothing above conflicts with it.

**6. Chunk streaming under fast movement.** `World.update()` refuses to rebuild the load
queue until the previous one drains (`this.loadHead >= this.loadQueue.length`), so a
sprinting or flying player keeps meshing chunks selected for a center they have already
left. Direction: keep the queue but add a generation counter and a cheap
"center moved by ≥ 1 chunk" test that *re-sorts in place* (the queue is already
distance-ordered, so re-scoring `d` against the new center and doing a bounded partial
sort is far cheaper than rebuilding), plus discard entries now outside `radius`.
`rebuildLoadQueue` also allocates two arrays and one object per candidate every rebuild;
replace with a preallocated `Int32Array` of packed chunk keys and a counting sort over
`d2` (≤ (2r+1)² entries, r = 5 → 121).

**7. Mesher border reads.** `boundGet` → `getBlockRaw` → `ensureData` means meshing one
chunk can generate up to four neighbours *and* mark meshed neighbours dirty, which is the
single worst frame spike in the current build. Direction: pass 2 of `buildChunkGeometry`
should read borders with `peekBlock` and store `UNKNOWN` (255) when the neighbour is
absent — the mesher already handles `UNKNOWN` correctly for both the face predicate and
AO (`OCCLUDES[UNKNOWN] = 1`, `localGet` maps 255 → −1). The chunk is then re-meshed for
free by the existing `markNeighborBorders` path once the neighbour's data really arrives.
Requires: `ensureData` must call `markNeighborBorders` even when `bulkPreparing` is
false-to-true transitions are involved, and the load queue must enqueue data-only
neighbours ahead of meshing. Both mesher bugs listed in §1.5 must be re-verified after
this change, because it makes `UNKNOWN` borders the common case instead of the rare one.
